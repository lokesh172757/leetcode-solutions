class Solution {
    public int longestBalanced(String s) {
        int n=s.length(),ans=0;
        for(int i=0;i<n;i++){
            int arr[]=new int[26];
            int max=0;
            for(int j=i;j<n;j++){
                if(n-i<=ans)return ans;
                arr[s.charAt(j)-'a']++;
                max=Math.max(max,arr[s.charAt(j)-'a']);
                boolean equal=true;
                for(int k=0;k<26;k++){
                    if(arr[k]!=0&&arr[k]!=max){
                        equal=false;
                        break;
                    }
                }
                if(equal)ans=Math.max(ans,j-i+1);
            }
        }
        return ans;
    }
}
