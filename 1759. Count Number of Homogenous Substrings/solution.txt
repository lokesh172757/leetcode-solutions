class Solution {
    static int mod = (int)1e9 + 7;
    public int countHomogenous(String s) {
        int i = 0, j = 0;
        long ans = 0;
        while (j < s.length()) {
            if (s.charAt(i) != s.charAt(j)) {
                long len = j - i;
                ans = (ans + (len * (len + 1)) / 2) % mod;
                i = j;
            }
            j++;
        }

        long len = j - i;
        ans = (ans + (len * (len + 1)) / 2) % mod;

        return (int) ans;
    }
}
