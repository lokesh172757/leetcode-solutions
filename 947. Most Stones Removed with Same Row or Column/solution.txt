class Solution {
    public static void dfs(int stones[][],int idx,boolean vis[]){
        vis[idx]=true;
        for(int j=0;j<stones.length;j++){
            if(!vis[j]&&(stones[idx][0]==stones[j][0]||stones[idx][1]==stones[j][1])){
                dfs(stones,j,vis);
            }
        }
    }
    public int removeStones(int[][] stones) {
        int n=stones.length;
        int group=0;
        boolean vis[]=new boolean[n];
        for(int i=0;i<n;i++){
            if(vis[i])continue;
            dfs(stones,i,vis);
            group++;
        }
        return n-group;
    }
}
