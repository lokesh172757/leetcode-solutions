class Solution {

    public static boolean check(long t,int mh,int time[]){
        long total=0;

        for(int i=0;i<time.length;i++){

            int x=1;
            long curtime=0;

            while(x<=mh){
                if(curtime + (long)time[i]*x <= t){
                    curtime += (long)time[i]*x;
                    x++;
                }
                else break;
            }

            total += x-1;

            if(total>=mh) return true;
        }

        return total>=mh;
    }

    public long minNumberOfSeconds(int mountainHeight, int[] workerTimes) {

        long st=1;
        long end=1000000L*((100001L*100000L)/2);
        long ans=end;

        while(st<=end){

            long mid=st+(end-st)/2;

            if(check(mid,mountainHeight,workerTimes)){
                ans=mid;
                end=mid-1;
            }
            else st=mid+1;
        }

        return ans;
    }
}
