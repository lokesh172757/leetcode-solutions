class Solution {
boolean canBeIncreasing(int[] nums) {
    int removed = 0;

    for (int i = 0; i < nums.length - 1; i++) {
        if (nums[i] >= nums[i + 1]) {
            removed++;
            if (removed > 1) return false;
            if (i > 0 && nums[i - 1] >= nums[i + 1]) {
                nums[i + 1] = nums[i];
            }
        }
    }
    return true;
}

}
