class Solution {
    public static int helper(int coins[][],int dp[][][],int i,int j,int m,int n,int count){
        if(i>=m||j>=n)return Integer.MIN_VALUE;
        if(i==m-1&&j==n-1){
            if(coins[i][j]<0&&count>0)return 0;
            return coins[i][j];
        }
        if(dp[i][j][count]!=Integer.MIN_VALUE)return dp[i][j][count];
        int ans1=0,ans2=0;
        int right=helper(coins,dp,i,j+1,m,n,count);
        ans1=right==Integer.MIN_VALUE?Integer.MIN_VALUE:right+coins[i][j];
        int down=helper(coins,dp,i+1,j,m,n,count);
        ans2=down==Integer.MIN_VALUE?Integer.MIN_VALUE:down+coins[i][j];
        
        if(count>0){
            ans1=Math.max(ans1,helper(coins,dp,i,j+1,m,n,count-1));
            ans2=Math.max(ans2,helper(coins,dp,i+1,j,m,n,count-1));
        }
        return dp[i][j][count]=Math.max(ans1,ans2);
    }
    public int maximumAmount(int[][] coins) {
        int m=coins.length,n=coins[0].length;
        int dp[][][]=new int[m][n][3];
        for(int ar[][]:dp){
            for(int a[]:ar)Arrays.fill(a,Integer.MIN_VALUE);
        }
        return helper(coins,dp,0,0,m,n,2);
    }
}
