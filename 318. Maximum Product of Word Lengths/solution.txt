class Solution {
    public int maxProduct(String[] words) {
        int n=words.length;
        int map[][]=new int[n][26];
        int count[][]=new int[n][2];
        for(int i=0;i<n;i++){
            count[i][0]=i;
            count[i][1]=words[i].length();
            for(int j=0;j<words[i].length();j++){
                char c=words[i].charAt(j);
                map[i][(int)(c-'a')]++;
            }
        }
        int ans=0;
        for(int i=n-1;i>0;i--){
            int id1=count[i][0];
            for(int j=i-1;j>=0;j--){
                int id2=count[j][0];
                boolean pair=true;
                for(int k=0;k<26;k++){
                    if(map[id1][k]!=0&&map[id2][k]!=0){
                        pair=false;
                        break;
                    }
                }
                if(pair)ans=Math.max(count[i][1]*count[j][1],ans);
            }
        }
        return ans;
    }
}
