class Solution {
    public int oddCells(int m, int n, int[][] indices) {
        int row[]=new int[m],col[]=new int[n];
        for(int i=0;i<indices.length;i++){
            row[indices[i][0]]++;
            col[indices[i][1]]++;
        }
        int oddRow=0,evenCol=0;
        for(int i=0;i<m;i++){
            if(row[i]%2==1)oddRow++;
        }
        for(int i=0;i<n;i++){
            if(col[i]%2==0)evenCol++;
        }
        return oddRow*evenCol+(m-oddRow)*(n-evenCol);
    }
}
