class Solution {
    public int missingInteger(int[] A) {
        int n = A.length;
        int arr[]=new int[51];

        for (int num : A)
            arr[num]++;
        int sum = A[0];

        for (int i = 1; i < n; i++) {
            if (A[i] == A[i - 1] + 1)
                sum += A[i];
            else break;
        }

        while (sum<arr.length&&arr[sum]!=0)
            sum++;

        return sum;
    }
}
