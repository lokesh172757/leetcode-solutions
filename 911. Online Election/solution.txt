class TopVotedCandidate {
    int times[],persons[];
    public TopVotedCandidate(int[] persons, int[] times) {
        this.times=times;
        int n=persons.length;
        this.persons=persons;
        int curr=-1,freq=0;
        HashMap<Integer,Integer>map=new HashMap<>();
        for(int i=0;i<n;i++){
            map.put(persons[i],map.getOrDefault(persons[i],0)+1);
            int pFreq=map.get(persons[i]);
            if(pFreq>=freq){
                freq=pFreq;
                curr=persons[i];
                
            }
            persons[i]=curr;
        }
    }
    public int binary(int t){
        int s=0,e=times.length-1,ans=0;
        while(s<=e){
            int mid=s+(e-s)/2;
            if(times[mid]==t)return mid;
            else if(times[mid]<t){
                ans=mid;
                s=mid+1;
            }
            else e=mid-1;
        }
        return ans;
    }
    
    public int q(int t) {
        return persons[binary(t)];
    }
}
