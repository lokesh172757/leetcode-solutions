class Solution {
    public int maximumLengthSubstring(String s) {
        int map[]=new int[26];
        int i=0,j=0,ans=0;
        while(j<s.length()){
            int c=(int)(s.charAt(j)-'a');
            map[c]++;
            while(map[c]>2&&i<j){
                int ci=(int)(s.charAt(i)-'a');
                map[ci]--;
                i++;
            }
            ans=Math.max(ans,j-i+1);
            j++;
        }
        return ans;
    }
}
