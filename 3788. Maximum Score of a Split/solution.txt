class Solution {
    public long maximumScore(int[] nums) {
        int n=nums.length;
        long max=Long.MIN_VALUE;
        long preSum=0;
        int suffMin[]=new int[n];
        int min=nums[n-1];
        suffMin[n-1]=min;
        for(int i=n-2;i>=0;i--){
            if(nums[i]<min){
                min=nums[i];
            }
            suffMin[i]=min;
        }
        for(int i=0;i<n-1;i++){
            preSum+=nums[i];
            long sum=preSum-suffMin[i+1];
            max=Math.max(max,sum);
        }
        return max;
    }
}
