class Solution {
    public int[] arrayRankTransform(int[] arr) {
        int n=arr.length;
        if(n==0)return arr;
        int nums[][]=new int[n][2];
        for(int i=0;i<n;i++){
            nums[i][0]=arr[i];
            nums[i][1]=i;
        }
        Arrays.sort(nums,Comparator.comparingInt(o->o[0]));
        int rank=1;
        arr[nums[0][1]]=1;
        for(int i=1;i<n;i++){
            if(nums[i][0]!=nums[i-1][0])rank++;
            arr[nums[i][1]]=rank;
        }
        return arr;
    }
}
