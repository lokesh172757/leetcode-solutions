class Solution {
    public int minOperations(String[] logs) {
        int len=0;
        for(int i=0;i<logs.length;i++){
            String s=logs[i];
            char ch1=s.charAt(0),ch2=s.charAt(1);
            if(ch1!='.')len++;
            else if(ch1=='.'&&ch2=='.'&&len>0)len--;

        }
        return len>0?len:0;
    }
}
