class Solution {
    static Boolean isSubsetSum(int arr[], int sum) {
        int m=arr.length,n=sum;
        boolean dp[][]=new boolean[m+1][n+1];
        for(int i=0;i<=m;i++){
            dp[i][0]=true;
        }   
        for(int i=1;i<=m;i++){
            for(int j=1;j<=n;j++){
                boolean ans1=dp[i-1][j];
                boolean ans2=false;
                if(arr[i-1]<=j){
                    ans2=dp[i-1][j-arr[i-1]];
                }
                dp[i][j]=ans1||ans2;
            }
        }
        return dp[m][n];
    }
}
