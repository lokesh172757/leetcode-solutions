class Solution {
    public long maximumHappinessSum(int[] happiness, int k) {
        Arrays.sort(happiness);
        int n=happiness.length,i=n-1,j=1;
        long ans=happiness[i];
        i--;
        k--;
        while(i>=0&&k>0){
            if(happiness[i]-j>0){
                ans+=happiness[i]-j;
            }
            else{
                break;
            }
            j++;
            i--;
            k--;
        }
        return ans;

    }
}
