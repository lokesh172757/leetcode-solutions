import java.util.*;

class Solution {
    static final int MOD = 1000000007;

    public static long helper(int n, int idx, ArrayList<Integer> l, long[][] dp) {
        if (n == 0) return 1;
        if (idx == l.size() || n < 0) return 0;

        if (dp[n][idx] != -1) return dp[n][idx];

        long take = helper(n - l.get(idx), idx + 1, l, dp) % MOD;
        long skip = helper(n, idx + 1, l, dp) % MOD;

        return dp[n][idx] = (take + skip) % MOD;
    }

    public int numberOfWays(int n, int x) {
        ArrayList<Integer> l = new ArrayList<>();
        int i = 1;
        while (Math.pow(i, x) <= n) {
            l.add((int) Math.pow(i, x));
            i++;
        }

        long[][] dp = new long[n + 1][l.size()];
        for (long[] row : dp) Arrays.fill(row, -1);

        return (int) (helper(n, 0, l, dp) % MOD);
    }
}
