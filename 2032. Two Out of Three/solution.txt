class Solution {
    public List<Integer> twoOutOfThree(int[] nums1, int[] nums2, int[] nums3) {
        int arr1[]=new int[101],arr2[]=new int[101],arr3[]=new int[101];
        for(int num:nums1)arr1[num]++;
        for(int num:nums2)arr2[num]++;
        for(int num:nums3)arr3[num]++;
        List<Integer>ans=new ArrayList<>();
        for(int i=1;i<=100;i++){
            if(arr1[i]!=0){
                if(arr2[i]!=0){
                    ans.add(i);
                    arr2[i]=0;
                }
                else if(arr3[i]!=0){
                    ans.add(i);
                    arr3[i]=0;
                }
            }
        }
        for(int i=1;i<=100;i++){
            if(arr2[i]!=0){
                if(arr3[i]!=0){
                    ans.add(i);
                }
            }
        }
        return ans;
    }
}
