class Solution {
    public boolean isBipartite(int V, int[][] edges) {
        ArrayList<Integer>graph[]=new ArrayList[V];
        for(int i=0;i<V;i++){
            graph[i]=new ArrayList<>();
        }
        for(int i=0;i<edges.length;i++){
            graph[edges[i][0]].add(edges[i][1]);
            graph[edges[i][1]].add(edges[i][0]);
        }
        int color[]=new int[V];
        Arrays.fill(color,-1);
        Queue<Integer>q=new LinkedList<>();
        for(int i=0;i<V;i++){
            if(color[i]==-1){
                color[i]=0;
                q.add(i);
                while(!q.isEmpty()){
                    int n=q.remove();
                    for(int j=0;j<graph[n].size();j++){
                        int d=graph[n].get(j);
                        if(color[d]==-1){
                            int ncolor=color[n]==0?1:0;
                            color[d]=ncolor;
                            q.add(d);
                        }
                        else if(color[d]==color[n]){
                            return false;
                        }
                    }
                    
                }
            }
        }
        return true;
        
    }
}
