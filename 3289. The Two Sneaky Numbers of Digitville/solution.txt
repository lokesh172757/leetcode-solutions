class Solution {
    public int[] getSneakyNumbers(int[] nums) {
        int arr[]=new int[nums.length];
        int ans[]=new int[2];
        int idx=0;
        for(int i=0;i<nums.length;i++){
            if(arr[nums[i]]==0){
                arr[nums[i]]++;
            }
            else{
                ans[idx]=nums[i];
                idx++;
            }
        }
        return ans;
    }
}
