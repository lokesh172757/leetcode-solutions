class Solution {

    public static boolean check(int nums[], int k, int mid) {

        int sum = 0;

        for(int i = 0; i < nums.length; i++) {

            if(nums[i] > mid)
                return false;

            if(sum + nums[i] > mid) {
                k--;
                sum = nums[i];
            } else {
                sum += nums[i];
            }
        }

        return k > 0;
    }

    public int splitArray(int[] nums, int k) {

        int min = 0;
        int max = 0;

        for(int num : nums) {
            min = Math.max(min, num);
            max += num;
        }

        int ans = max;

        while(min <= max) {

            int mid = min + (max - min) / 2;

            if(check(nums, k, mid)) {
                ans = mid;
                max = mid - 1;
            } else {
                min = mid + 1;
            }
        }

        return ans;
    }
}
