class Solution {
    public int buyChoco(int[] nums, int money) {
        int a=Integer.MAX_VALUE,b=Integer.MAX_VALUE;
        for(int i=0;i<nums.length;i++){
            if(nums[i]<a){
                b=a;
                a=nums[i];
            }
            else if(nums[i]<b)b=nums[i];
        }
        if(a+b>money)return money;
        return money-a-b;
    }
}
