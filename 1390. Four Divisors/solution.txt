class Solution {
    public static int check(int n){
        int div=0,sum=n+1;
        for(int i=2;i<=n/2;i++){
            if(n%i==0){
                div++;
                sum+=i;
            }
            if(div>2)return 0;
        }
        return div==2?sum:0;
    }
    public int sumFourDivisors(int[] nums) {
        int ans=0;
        for(int i=0;i<nums.length;i++){
            if(nums[i]<6)continue;
            ans+=check(nums[i]);
        }
        return ans;
    }
}
