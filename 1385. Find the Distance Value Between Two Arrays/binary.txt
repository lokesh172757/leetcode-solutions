class Solution {
    public static boolean find(int arr[],int ri,int re){
        int s=0,e=arr.length-1;
        while(s<=e){
            int mid=(s+e)/2;
            if(arr[mid]>=ri&&arr[mid]<=re)return true;
            else if(arr[mid]>re)e=mid-1;
            else s=mid+1;
        }
        return false;
    }
    public int findTheDistanceValue(int[] arr1, int[] arr2, int d) {
        Arrays.sort(arr2);
        int n=arr1.length,ans=0;
        for(int i=0;i<n;i++){
            if(!find(arr2,arr1[i]-d,arr1[i]+d))ans++;
        }
        return ans;
    }
}
