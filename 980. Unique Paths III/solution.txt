class Solution {
    public static int helper(int[][] grid,int i,int j,int er,int ec,int m,int n,int count){
        if(i>=m||j>=n||i<0||j<0||grid[i][j]==-1){
            return 0;
        }
        else if(i==er&&j==ec&&count==0){
            return 1;
        }
        int val=grid[i][j];
        grid[i][j]=-1;
        int ans=helper(grid,i+1,j,er,ec,m,n,count-1)+
        helper(grid,i-1,j,er,ec,m,n,count-1)+
        helper(grid,i,j+1,er,ec,m,n,count-1)+
        helper(grid,i,j-1,er,ec,m,n,count-1);
        grid[i][j]=val;
        return ans;
        
    }
    public int uniquePathsIII(int[][] grid) {
        int m=grid.length,n=grid[0].length;
        int sr=-1,sc=-1,er=-1,ec=-1,count=0;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]==-1){
                    continue;
                }
                if(grid[i][j]==1){
                    sr=i;
                    sc=j;
                }
                else if(grid[i][j]==2){
                    er=i;
                    ec=j;
                }
                count++;
            }
        }
        return helper(grid,sr,sc,er,ec,m,n,count-1);
        
        
    }
}
