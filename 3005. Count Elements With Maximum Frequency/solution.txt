class Solution {
    public int maxFrequencyElements(int[] nums) {
        int arr[]=new int[101];
        int mfreq=0;
        for(int i=0;i<nums.length;i++){
            arr[nums[i]]++;
            if(mfreq<arr[nums[i]]){
                mfreq=arr[nums[i]];
            }
        }
        int ans=0;
        for(int i=1;i<101;i++){
            if(arr[i]==mfreq){
                ans+=mfreq;
            }
        }
        return ans;
    }
}
