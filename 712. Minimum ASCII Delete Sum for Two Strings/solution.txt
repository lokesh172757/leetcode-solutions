class Solution {
    public int minimumDeleteSum(String s1, String s2) {
        int m=s1.length();
        int n=s2.length();
        int total=0;
        for(int i=0;i<m;i++){
            total+=(int)s1.charAt(i);
        }
        for(int i=0;i<n;i++){
            total+=(int)s2.charAt(i);
        }
        int dp[]=new int[n+1];
        for(int i=1;i<=m;i++){
            int prevDiag=0;
            for(int j=1;j<=n;j++){
                int temp=dp[j];
                if(s1.charAt(i-1)==s2.charAt(j-1)){
                    dp[j]=prevDiag+(int)s1.charAt(i-1);
                }
                else{
                    dp[j]=Math.max(dp[j],dp[j-1]);
                }
                prevDiag=temp;
            }
        }
        return total-2*dp[n];
    }
}
