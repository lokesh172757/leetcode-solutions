class Solution {
    public int smallestRangeI(int[] nums, int k) {
        int n=nums.length;
        if(n==1) return 0;
        int s=Integer.MAX_VALUE,m=-1;
        for(int i=0;i<n;i++){
            s=Math.min(s,nums[i]);
            m=Math.max(m,nums[i]);
        }
        if(m-s<=2*k)return 0;
        return m-s-2*k;
    }
}
