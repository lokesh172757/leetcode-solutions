class Solution {
    public List<Integer> survivedRobotsHealths(int[] positions, int[] healths, String directions) {
        int n=positions.length;
        List<Integer>ans=new ArrayList<>();
        if(n==1){
            ans.add(healths[0]);
            return ans;
        }
        ArrayDeque<Integer>st=new ArrayDeque<>();
        int arr[][]=new int[n][3];//pos,hel,dir->0=L,1=R
        HashMap<Integer,Integer>map=new HashMap<>();
        for(int i=0;i<n;i++){
            arr[i][0]=positions[i];
            arr[i][1]=healths[i];
            arr[i][2]=directions.charAt(i)=='L'?0:1;
        }
        Arrays.sort(arr,Comparator.comparingInt(o->o[0]));
        for(int i=0;i<n;i++){
            map.put(arr[i][0],i);
        }
        for(int i=0;i<n;i++){
            if(arr[i][2]==1)st.push(i);
            else{
                if(st.isEmpty())st.push(i);
                else{
                    int lastIdx=st.peek();
                    if(arr[lastIdx][2]==0)st.push(i);
                    else{
                        boolean pushI=true;
                        while(!st.isEmpty()&&arr[lastIdx][2]==1){
                            if(arr[lastIdx][1]==arr[i][1]){
                                st.pop();
                                arr[lastIdx][1]=0;
                                arr[i][1]=0;
                                pushI=false;
                                break;
                            }
                            else if(arr[lastIdx][1]>arr[i][1]){
                                arr[lastIdx][1]--;
                                arr[i][1]=0;
                                pushI=false;
                                break;
                            }
                            else{
                                st.pop();
                                arr[lastIdx][1]=0;
                                arr[i][1]--;
                                if(!st.isEmpty())lastIdx=st.peek();
                            }
                        }
                        if(pushI)st.push(i);
                    }
                }
                
            }
        }
        if(st.isEmpty())return ans;
        for(int i=0;i<n;i++){
            int idx=map.get(positions[i]);
            if(arr[idx][1]!=0){
                ans.add(arr[idx][1]);
            }
        }
        return ans;
    }   
}
