class Solution {
    public int matrixSum(int[][] nums) {
        for(int ar[]:nums){
            Arrays.sort(ar);
        }
        int ans=0;
        for(int j=nums[0].length-1;j>=0;j--){
            int max=-1;
            for(int i=0;i<nums.length;i++){
                max=Math.max(nums[i][j],max);
            }
            ans+=max;
        }
        return ans;
    }
}
