class Solution {
    public int countSpecialIntegers(int[] nums) {
        int arr[]=new int[101];
        int ans=0;
        for(int i=0;i<nums.length;i++){
            int num=nums[i];
            if(arr[num]==0){
                ans++;
                arr[num]++;
            }
            else if(arr[num]==1){
                ans--;
                arr[num]=-1;
            }
            while(i<nums.length-1&&nums[i+1]==num)i++;
        }
        return ans;
    }
}
