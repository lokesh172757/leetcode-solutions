class Solution {
    public boolean canThreePartsEqualSum(int[] arr) {
        int n=arr.length,sum=0;
        for(int i=0;i<n;i++){
            sum+=arr[i];
        }
        
        if(sum%3!=0)return false;
        int part=sum/3,curr=0,div=0;
        for(int i=0;i<n;i++){
            curr+=arr[i];
            if(curr==part){
                curr=0;
                div++;
            }
            if(div==3)return true;
        }
        return curr==0&&div==3;
    }
}
