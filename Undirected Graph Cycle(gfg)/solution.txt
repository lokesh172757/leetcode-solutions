class Solution {
    public static boolean isCyclicUtil(ArrayList<Integer>graph[],boolean vis[],int par,int src){
        vis[src]=true;
        for(int i=0;i<graph[src].size();i++){
            int n=graph[src].get(i);
            if(vis[n]&&n!=par){
                return true;
            }
            else if(!vis[n]){
                if(isCyclicUtil(graph,vis,src,n)){
                    return true;
                }
            }
            
        }
        return false;
    }
    public boolean isCycle(int V, int[][] edges) {
        ArrayList<Integer>graph[]=new ArrayList[V];
        boolean vis[]=new boolean[V];
        for(int i=0;i<V;i++){
            graph[i]=new ArrayList<>();
        }
        for(int i=0;i<edges.length;i++){
            graph[edges[i][0]].add(edges[i][1]);
            graph[edges[i][1]].add(edges[i][0]);
        }
        for(int i=0;i<V;i++){
            if(!vis[i]){
                if(isCyclicUtil(graph,vis,-1,i)){
                    return true;
                }
            }
        }
        return false;
        
    }
}
