class Solution {

    static class Node {
        Node child[] = new Node[26];
        boolean eow = false;
    }

    public static void makeTrie(String s, Node root) {
        for (int i = 0; i < s.length(); i++) {
            int idx = s.charAt(i) - 'a';
            if (root.child[idx] == null)
                root.child[idx] = new Node();
            root = root.child[idx];
        }
        root.eow = true;
    }

    public static String longest(Node root, String curr) {
        String ans = curr;

        for (int i = 0; i < 26; i++) {
            if (root.child[i] != null && root.child[i].eow) {
                String rec = longest(root.child[i], curr + (char)(i + 'a'));

                if (rec.length() > ans.length() || 
                   (rec.length() == ans.length() && rec.compareTo(ans) < 0)) {
                    ans = rec;
                }
            }
        }
        return ans;
    }

    public String longestWord(String[] words) {
        Node root = new Node();

        for (String word : words) {
            makeTrie(word, root);
        }

        return longest(root, "");
    }
}
