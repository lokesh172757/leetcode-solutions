class Solution {
    public boolean canPartitionGrid(int[][] grid) {
        int m=grid.length,n=grid[0].length;
        long row[]=new long[m],col[]=new long[n];
        long total=0;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                total+=grid[i][j];
                col[j]+=grid[i][j];
                row[i]+=grid[i][j];
            }
        }
        if(total%2==1)return false;
        long rowSum=0;
        for(int i=0;i<m-1;i++){
            rowSum+=row[i];
            if(rowSum==total/2)return true;
        }
        long colSum=0;
        for(int i=0;i<n-1;i++){
            colSum+=col[i];
            if(colSum==total/2)return true;
        }
        return false;
    }
}
