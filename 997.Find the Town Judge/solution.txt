class Solution {
    public int findJudge(int n, int[][] trust) {
        boolean arr[]=new boolean[n+1];
        int trcnt[]=new int [n+1];
        for(int i=0;i<trust.length;i++){
            arr[trust[i][0]]=true;
            trcnt[trust[i][1]]++;
        }
        for(int i=1;i<=n;i++){
            if(!arr[i]&&trcnt[i]==n-1){
                return i;
            }
        }
        return -1;
    }
}
