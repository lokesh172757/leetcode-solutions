class Solution {
    public String maximumOddBinaryNumber(String s) {
        int one=0,zero=0;
        for(int i=0;i<s.length();i++){
            if(s.charAt(i)=='0'){
                zero++;
            }
            else{
                one++;
            }
        }
        StringBuilder sb=new StringBuilder("");
        if(one>0){
            sb.append('1');
            one--;
        }
        while(zero>0){
            sb.append('0');
            zero--;
        }
        while(one>0){
            sb.append('1');
            one--;
        }
        return sb.reverse().toString();
    }
}
