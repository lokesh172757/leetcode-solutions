class Solution {
    public int countSubmatrices(int[][] grid, int k) {
        int m=grid.length,n=grid[0].length;
        for(int i=0;i<m;i++){
            for(int j=1;j<n;j++){
                grid[i][j]+=grid[i][j-1];
            }
        }
        int ans=0;
        for(int j=0;j<n;j++){
            int sum=0;
            for(int i=0;i<m;i++){
                sum+=grid[i][j];
                if(sum<=k)ans++;
                else break;
            }
        }
        return ans;
    }
}
