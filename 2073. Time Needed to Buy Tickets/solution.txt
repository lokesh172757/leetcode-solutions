class Solution {
    static class Pair{
        int i,t;
        public Pair(int i,int t){
            this.i=i;
            this.t=t;
        }
    }
    public int timeRequiredToBuy(int[] tickets, int k) {
        int n=tickets.length;
        int t=0;   
        Queue<Pair>q=new LinkedList<>();
        for(int i=0;i<n;i++){
            q.add(new Pair(i,tickets[i]));
        }
    
        while(!q.isEmpty()){
            Pair p=q.remove();
            t++;
            if(p.t-1>0)q.add(new Pair(p.i,p.t-1));
            else if(p.i==k) break;
        }
        return t;
    }
}
