class Solution {
    public boolean isBipartite(int[][] graph) {
        int n = graph.length;
        int[] color = new int[n];
        Arrays.fill(color, -1);

        Queue<Integer> q = new LinkedList<>();

        for (int i = 0; i < n; i++) {
            if (color[i] == -1) {
                q.add(i);
                color[i] = 0;

                while (!q.isEmpty()) {
                    int curr = q.remove();
                    for (int dest : graph[curr]) {
                        if (color[dest] == -1) {
                            color[dest] = 1 - color[curr];
                            q.add(dest);
                        } else if (color[dest] == color[curr]) {
                            return false;
                        }
                    }
                }
            }
        }
        return true;
    }
}
