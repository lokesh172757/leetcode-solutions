class Solution {
    public int minimumCost(int[] nums) {
        int f = Integer.MAX_VALUE,s= Integer.MAX_VALUE;
        for (int i = 1; i < nums.length; i++) {
            int x = nums[i];
            if (x < f) {
                s = f;
                f = x;
            } else if (x<s){
                s=x;
            }
        }
        return nums[0]+f+s;
    }
}
