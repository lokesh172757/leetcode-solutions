class Solution {
    public int edgeScore(int[] edges) {
        int n=edges.length;
        long arr[]=new long[n];
        for(int i=0;i<n;i++){
            arr[edges[i]]+=i;
        }
        int ans=-1;
        long max=0;
        for(int i=0;i<n;i++){
            if(arr[i]>max){
                max=arr[i];
                ans=i;
            }
        }
        return ans;
    }
}
