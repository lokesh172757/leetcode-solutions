class Solution {
    public int maxFreqSum(String s) {
        int arr[]=new int[26];
        for(int i=0;i<s.length();i++){
            int idx=(int)(s.charAt(i)-'a');
            arr[idx]++;
        }
        int v=0,c=0;
        for(int i=0;i<26;i++){
            if(i==0||i==4||i==8||i==14||i==20){
                if(arr[i]>v){
                    v=arr[i];
                }
            }
            else{
                if(arr[i]>c){
                    c=arr[i];
                }
            }
        }
        return v+c;
    }
}
