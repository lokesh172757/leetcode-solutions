class Solution {
    public int findMinArrowShots(int[][] points) {
        Arrays.sort(points,Comparator.comparingInt(o->o[0]));
        int ans=1;
        int st=points[0][0],end=points[0][1];
        for(int i=1;i<points.length;i++){
            int s=points[i][0],e=points[i][1];
            if(s<=end){
                st=Math.max(st,s);
                end=Math.min(end,e);
            }
            else{
                ans++;
                st=s;
                end=e;
            }
        }
        return ans;
    }
}
