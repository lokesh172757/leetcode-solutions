class Solution {
    public int maxDepth(String s) {
        int n=s.length(),open=0,ans=0;
        for(int i=0;i<n;i++){
            char ch=s.charAt(i);
            if(ch=='('){
                open++;
                ans=Math.max(open,ans);
            }
            else if(ch==')')open--;
        }
        return ans;
    }
}
