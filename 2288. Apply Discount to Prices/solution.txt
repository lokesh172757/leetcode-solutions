class Solution {
    public String discountPrices(String sentence, int discount) {
        String arr[]=sentence.split(" ");
        StringBuilder sb=new StringBuilder("");
        for(int i=0;i<arr.length;i++){
            if(arr[i].charAt(0)=='$'&&arr[i].length()>1){
                boolean repeate=false;
                for(int j=1;j<arr[i].length();j++){
                    if(arr[i].charAt(j)=='$'||(arr[i].charAt(j)>='a'&&arr[i].charAt(j)<='z')){
                        repeate=true;
                        break;
                    }
                }
                if(repeate){
                    sb.append(arr[i]+" ");
                }
                else{
                    long num = Long.parseLong(arr[i].substring(1));

                    String changed=String.format("%.2f",num-(num/100.0)*discount);
                    sb.append("$"+changed+" ");
                }
                
            }
            else{
                sb.append(arr[i]+" ");
            }
        }
        return sb.toString().trim();
    }
}
