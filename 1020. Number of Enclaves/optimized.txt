class Solution {
    public static void makeZero(int grid[][],int i,int j,int m,int n){
        if(i<0||i>=m||j<0||j>=n||grid[i][j]==0)return;
        grid[i][j]=0;
        makeZero(grid,i+1,j,m,n);
        makeZero(grid,i-1,j,m,n);
        makeZero(grid,i,j+1,m,n);
        makeZero(grid,i,j-1,m,n);
    }
    public int numEnclaves(int[][] grid) {
        int m=grid.length,n=grid[0].length;
        for(int i=0;i<m;i++){
            if(grid[i][0]==1){
                makeZero(grid,i,0,m,n);
            }
            if(grid[i][n-1]==1){
                makeZero(grid,i,n-1,m,n);
            }
        }
        for(int i=1;i<n-1;i++){
            if(grid[0][i]==1){
                makeZero(grid,0,i,m,n);
            }
            if(grid[m-1][i]==1){
                makeZero(grid,m-1,i,m,n);
            }
        }
        int ans=0;
        for(int i=1;i<m-1;i++){
            for(int j=1;j<n-1;j++){
                if(grid[i][j]==1)ans++;
            }
        }
        return ans;
    }
}
