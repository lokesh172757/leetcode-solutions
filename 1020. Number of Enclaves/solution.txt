class Solution {
    public static int checkBoundary(int grid[][],int i,int j,int m,int n){
        if(i<0||i>=m||j<0||j>=n){
            return-1;
        }
        else if(grid[i][j]==0)return 0;
        grid[i][j]=0;
        int a=checkBoundary(grid,i,j+1,m,n);
        int b=checkBoundary(grid,i,j-1,m,n);
        int c=checkBoundary(grid,i+1,j,m,n);
        int d=checkBoundary(grid,i-1,j,m,n);
        if(a==-1||b==-1||c==-1||d==-1)return -1;
        else{
            return a+b+c+d+1;
        }

    }
    public int numEnclaves(int[][] grid) {
        int m=grid.length,n=grid[0].length,ans=0;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]==1){
                    int count=checkBoundary(grid,i,j,m,n);
                    if(count!=-1)ans+=count;
                }
            }
        }
        return ans;
    }
}
