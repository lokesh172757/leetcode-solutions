class Solution {
    public int numSteps(String s) {
        int carry=0,ans=0;
        for(int i=s.length()-1;i>=0;i--){
            if(i==0&&carry==0)break;
            char ch=s.charAt(i);
            if(ch=='1'){
                if(carry==0){
                    carry=1;
                    ans++;
                }
            }
            else{
                if(carry==1){
                    ans++;
                }
            }
            ans++;
        }
        return ans;
    }
}
