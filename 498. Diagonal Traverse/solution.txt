class Solution {
    public int[] findDiagonalOrder(int[][] mat) {
        int m = mat.length, n = mat[0].length;
        int ans[] = new int[m * n];
        int i = 0, j = 0, k = 0;

        while (k < m * n) {
            while (i >= 0 && j < n) {
                ans[k++] = mat[i][j];
                i--;
                j++;
            }
            if (j >= n) {
                j = n - 1;
                i += 2;
            } else {
                i = 0;
            }
            if (k >= m * n) break;

            while (j >= 0 && i < m) {
                ans[k++] = mat[i][j];
                i++;
                j--;
            }
            if (i >= m) {
                i = m - 1;
                j += 2;
            } else {
                j = 0;
            }
        }
        return ans;
    }
}
