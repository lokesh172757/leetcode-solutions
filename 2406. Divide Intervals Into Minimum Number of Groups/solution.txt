class Solution {
    public int minGroups(int[][] intervals) {
        int max=0;
        for(int ar[]:intervals){
            max=Math.max(max,ar[1]);
        }
        int pre[]=new int[max+1];

        for(int ar[]:intervals){
            pre[ar[0]]++;
            if(ar[1]+1<=max)pre[ar[1]+1]--;
        }
        int ans=1;
        for(int i=1;i<=max;i++){
            pre[i]+=pre[i-1];
            ans=Math.max(ans,pre[i]);
        }
        return ans;
    }
}
