class Solution {
    public int minNumberOfFrogs(String croakOfFrogs) {
        int[] cnt = new int[5]; // c r o a k
        int active = 0, ans = 0;

        for (char ch : croakOfFrogs.toCharArray()) {
            if (ch == 'c') {
                cnt[0]++;
                active++;
                ans = Math.max(ans, active);
            } else if (ch == 'r') {
                if (cnt[0] <= cnt[1]) return -1;
                cnt[1]++;
            } else if (ch == 'o') {
                if (cnt[1] <= cnt[2]) return -1;
                cnt[2]++;
            } else if (ch == 'a') {
                if (cnt[2] <= cnt[3]) return -1;
                cnt[3]++;
            } else if (ch == 'k') {
                if (cnt[3] <= cnt[4]) return -1;
                cnt[4]++;
                active--;
            } else {
                return -1;
            }
        }
        for (int i = 1; i < 5; i++) {
            if (cnt[i] != cnt[0]) return -1;
        }

        return ans;
    }
}
