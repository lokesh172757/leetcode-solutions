class Solution {
    public int firstStableIndex(int[] nums, int k) {
        int n=nums.length,max=0,min=Integer.MAX_VALUE;
        int ans=n;
        int score[]=new int[n];
        for(int i=0;i<n;i++){
            max=Math.max(nums[i],max);
            score[i]=max;
        }
        for(int i=n-1;i>=0;i--){
            min=Math.min(nums[i],min);
            score[i]-=min;
            if(score[i]<=k)ans=i;
        }
        return ans==n?-1:ans;
    }
}
