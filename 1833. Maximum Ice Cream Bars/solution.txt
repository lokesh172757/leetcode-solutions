class Solution {
    public int maxIceCream(int[] costs, int coins) {
        int ans=0,n=costs.length;
        Arrays.sort(costs);
        for(int i=0;i<n;i++){
            if(coins>=costs[i]){
                coins-=costs[i];
                ans++;
            }
            else break;
        }
        return ans;
    }
}
