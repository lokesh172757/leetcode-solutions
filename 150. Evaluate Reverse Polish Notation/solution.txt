class Solution {
    public int evalRPN(String[] tokens) {
        int n=tokens.length;
        Stack<Integer>st=new Stack<>();
        for(int i=0;i<n;i++){
            char ch=tokens[i].charAt(0);
            if(tokens[i].length()==1&&(ch=='+'||ch=='*'||ch=='/'||ch=='-')){
                int b=st.pop(),a=st.pop();
                if(ch=='+')st.push(a+b);
                else if(ch=='-')st.push(a-b);
                else if(ch=='*')st.push(a*b);
                else{
                    st.push(a/b);
                }
            }
            else{
                int num=Integer.parseInt(tokens[i]);
                st.push(num);
            }
        }
        return st.pop();
    }
}
