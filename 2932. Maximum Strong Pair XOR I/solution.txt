class Solution {
    public int maximumStrongPairXor(int[] nums) {
        int n=nums.length;
        Arrays.sort(nums);
        int ans=0;
        for(int i=0;i<n-1;i++){
            int j=i+1;
            while(j<n&&nums[j]-nums[i]<=nums[i]){
                ans=Math.max(ans,nums[i]^nums[j]);
                j++;
            }
        }
        return ans;
    }
}
