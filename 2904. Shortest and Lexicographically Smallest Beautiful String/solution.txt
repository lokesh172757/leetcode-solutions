class Solution {
    public String shortestBeautifulSubstring(String s, int k) {
        int i=0,j=0,n=s.length();
        int len=n+1,st=-1,e=-1,count=0;
        while(j<n){
            if(s.charAt(j)=='1')count++;
            while((count>k||s.charAt(i)=='0')&&i<j){
                if(s.charAt(i)=='1')count--;
                i++;
            }
            if(count==k){
                if(j-i+1<len){
                    st=i;
                    e=j;
                    len=j-i+1;
                }
                else if(j-i+1==len){
                    if(s.substring(i,j+1).compareTo(s.substring(st,e+1)) < 0){
                        st=i;
                        e=j;
                    }
                }
            }
            j++;
        }
        if(st==-1&&e==-1)return "";
        return s.substring(st,e+1);
    }
}
