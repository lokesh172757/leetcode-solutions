class Solution {
    public static int helper(int[][] matrix,int dp[][],int i,int j,int src){
        int m=matrix.length,n=matrix[0].length;
        if(i<0||i>=m||j<0||j>=n||matrix[i][j]<=src)return 0;
        if(dp[i][j]!=0)return dp[i][j];
        int dir[][]={{-1,+1,0,0},
                     {0,0,+1,-1}
                    };
        int ans=0;
        for(int k=0;k<4;k++){
            ans=Math.max(ans,helper(matrix,dp,i+dir[0][k],j+dir[1][k],matrix[i][j]));
        }
        return dp[i][j]=ans+1;
    }
    public int longestIncreasingPath(int[][] matrix) {
        int ans=0,m=matrix.length,n=matrix[0].length;
        int dp[][]=new int[m][n];

        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(dp[i][j]==0){
                    ans=Math.max(ans,helper(matrix,dp,i,j,-1));
                }
            }
        }
        return ans;
    }
}
