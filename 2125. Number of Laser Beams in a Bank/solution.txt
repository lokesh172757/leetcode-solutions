class Solution {
    public int numberOfBeams(String[] bank) {
        int n=bank.length;
        int last=0,ans=0;
        for(int i=0;i<n;i++){
            int d=0;
            for(int j=0;j<bank[i].length();j++){
                if(bank[i].charAt(j)=='1'){
                    d++;
                }
            }
            if(d!=0){
                ans+=last*d;
                last=d;
            }
        }
        return ans;
    }
}
