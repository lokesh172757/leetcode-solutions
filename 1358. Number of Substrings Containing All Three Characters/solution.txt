class Solution {
    public int numberOfSubstrings(String s) {
        int freq[] = new int[3];
        int ans=0,count=0,i=0,j=0,n=s.length();
        while(j<n){
            char ch = s.charAt(j);
            freq[ch - 'a']++;
            if(freq[ch-'a'] == 1){
                count++;
            }
            while(count==3){
                ans += (n -j);
                char c = s.charAt(i);
                freq[c - 'a']--;
                if(freq[c-'a'] == 0){
                    count--;
                }   
                i++;
            }
            j++;
        }
        return ans;
    }
}
