class Solution {
    public int findSmallestInteger(int[] nums, int k) {
        int n=nums.length;
        int val[]=new int[n];
        for(int num:nums){
            int v=((num%k)+k)%k;
            if(v<n)val[v]++;
        }
        for(int i=0;i<n;i++){
            if(val[i]==0)return i;
            val[i]--;
            if(val[i]!=0&&i+k<n){
                val[i+k]+=val[i];
            }
        }
        return n;
    }
}
