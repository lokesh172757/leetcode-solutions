class Solution {
    public int rob(int[] nums) {
        int n = nums.length;
        int[] dp = new int[n];
        Arrays.fill(dp, -1);
        return helper(nums, 0, dp);
    }

    private int helper(int[] nums, int idx, int[] dp) {
        if (idx >= nums.length) {
            return 0;
        }
        if (dp[idx] != -1) {
            return dp[idx];
        }
        int ans1 = helper(nums, idx + 1, dp);
        int ans2 = nums[idx] + helper(nums, idx + 2, dp);
        return dp[idx] = Math.max(ans1, ans2);
    }
}
