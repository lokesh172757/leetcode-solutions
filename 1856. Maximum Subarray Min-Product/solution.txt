class Solution {
    public int maxSumMinProduct(int[] arr) {
        int n = arr.length;
        int mod = (int) 1e9 + 7;

        int[] left = new int[n];
        int[] right = new int[n];

        long[] prefix = new long[n];
        prefix[0] = arr[0];
        for (int i = 1; i < n; i++) {
            prefix[i] = prefix[i - 1] + arr[i];
        }

        Stack<Integer> st = new Stack<>();

        // previous smaller element
        for (int i = 0; i < n; i++) {
            while (!st.isEmpty() && arr[st.peek()] >= arr[i]) {
                st.pop();
            }
            left[i] = st.isEmpty() ? 0 : st.peek() + 1;
            st.push(i);
        }

        st.clear();

        // next smaller element
        for (int i = n - 1; i >= 0; i--) {
            while (!st.isEmpty() && arr[st.peek()] > arr[i]) {
                st.pop();
            }
            right[i] = st.isEmpty() ? n - 1 : st.peek() - 1;
            st.push(i);
        }

        long ans = 0;

        for (int i = 0; i < n; i++) {
            long sum = prefix[right[i]] - (left[i] > 0 ? prefix[left[i] - 1] : 0);
            ans = Math.max(ans, sum * arr[i]);
        }

        return (int) (ans % mod);
    }
}
