class Solution {
    public static int find(int i,int j,char grid[][],int dp[][][],int open){
        if(i>=grid.length||j>=grid[0].length)return -1;
        if(grid[i][j]=='(')open++;
        else open--;
        if(open<0)return-1;
        if(dp[i][j][open]!=0)return dp[i][j][open];
        if(i==grid.length-1&&j==grid[0].length-1&&open==0)return 1;
        if(find(i,j+1,grid,dp,open)==1)return dp[i][j][open]=1;
        else if(find(i+1,j,grid,dp,open)==1)return dp[i][j][open]=1;
        return dp[i][j][open]=-1;
    }
    public boolean hasValidPath(char[][] grid) {
        int m=grid.length,n=grid[0].length;
        int dp[][][]=new int[m][n][m+n];
        return find(0,0,grid,dp,0)==1;
    }
}
