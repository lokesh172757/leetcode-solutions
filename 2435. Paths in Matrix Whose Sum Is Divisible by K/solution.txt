class Solution {
    public static int path(int i, int j, int rem, int grid[][], int k, int mod, int dp[][][], int m, int n) {
        if (i >= m || j >= n) return 0;

        if (i == m - 1 && j == n - 1) {
            if ((rem + grid[i][j]) % k == 0) return 1;
            return 0;
        }

        if (dp[i][j][rem] != -1) return dp[i][j][rem];

        int newRem = (rem + grid[i][j]) % k;

        long down = path(i + 1, j, newRem, grid, k, mod, dp, m, n);
        long right = path(i, j + 1, newRem, grid, k, mod, dp, m, n);

        return dp[i][j][rem] = (int) ((down + right) % mod);
    }

    public int numberOfPaths(int[][] grid, int k) {
        int mod = (int) 1e9 + 7;
        int m = grid.length, n = grid[0].length;

        int dp[][][] = new int[m][n][k];
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                for (int r = 0; r < k; r++)
                    dp[i][j][r] = -1;

        return path(0, 0, 0, grid, k, mod, dp, m, n);
    }
}
