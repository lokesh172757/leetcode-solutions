class Solution {
    public int longestCommonSubstr(String s1, String s2) {
        int m=s1.length(),n=s2.length();
        int dp[]=new int[n+1];
        int max=0;
        for(int i=1;i<=m;i++){
            int prevDiag=0;
            for(int j=1;j<=n;j++){
                int temp=dp[j];
                if(s1.charAt(i-1)==s2.charAt(j-1)){
                    dp[j]=prevDiag+1;
                    max=Math.max(max,dp[j]);
                }
                else{
                    dp[j]=0;
                }
                prevDiag=temp;
            }
        }
        return max;
        
    }
}
