class Solution {
    public int countTriples(int n) {
        int ans=0;
        for(int i=n;i>=3;i--){
            for(int j=i-2;j>=1;j--){
                int c2=i*i,a2=j*j;
                double b=Math.sqrt(c2-a2);
                if(b==(int)b&&b<i&&b>j){
                    ans+=2;
                }
            }
        }
        return ans;
    }
}
