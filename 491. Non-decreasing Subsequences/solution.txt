class Solution {
    public static void find(HashSet<List<Integer>>set,List<Integer>curr,int nums[],int i,int last){
        if(i>=nums.length){
            if(curr.size()>1)set.add(new ArrayList(curr));
            return;
        }
        find(set,curr,nums,i+1,last);
        if(nums[i]>=last){
            curr.add(nums[i]);
            find(set,curr,nums,i+1,nums[i]);
            curr.remove(curr.size()-1);
        }
    }
    public List<List<Integer>> findSubsequences(int[] nums) {
        List<List<Integer>>ans=new ArrayList<>();
        HashSet<List<Integer>>set=new HashSet<>();
        List<Integer>curr=new ArrayList<>();
        find(set,curr,nums,0,-101);
        for(List<Integer>l:set)ans.add(l);
        return ans;   
    }
}
