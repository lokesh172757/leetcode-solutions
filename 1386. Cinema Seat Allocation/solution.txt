class Solution {
    public int maxNumberOfFamilies(int n, int[][] reservedSeats) {
        int ans = 2 * n;
        HashMap<Integer, boolean[]> map = new HashMap<>();

        for (int[] seat : reservedSeats) {
            int row = seat[0];
            int col = seat[1];

            if (col >= 2 && col <= 9) {
                map.putIfAbsent(row, new boolean[3]);

                if (col >= 2 && col <= 5)
                    map.get(row)[0] = true;

                if (col >= 4 && col <= 7)
                    map.get(row)[1] = true;

                if (col >= 6 && col <= 9)
                    map.get(row)[2] = true;
            }
        }

        for (boolean[] arr : map.values()) {
            if (!arr[0] && !arr[2])
                continue;

            if (!arr[0] || !arr[1] || !arr[2])
                ans--;

            else
                ans -= 2;
        }

        return ans;
    }
}
