class Solution {
    public int countCommas(int n) {
        int ans=0,next=1000000,comma=1,curr=1000;
        while(curr<=n){
            if(n<next){
                ans=ans+(n-curr+1)*comma;
            }
            else{
                ans=ans+1+(n-curr+1)*comma;
            }
            curr=next+1;
            next*=1000;
            comma++;
        }
        return ans;
    }
}
