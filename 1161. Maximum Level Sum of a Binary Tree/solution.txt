class Solution {
    public int maxLevelSum(TreeNode root) {
        Queue<TreeNode> q = new LinkedList<>();
        q.add(root);
        q.add(null);
        int max = Integer.MIN_VALUE, level = 1;
        int curr = 0, ans = 1;

        while (!q.isEmpty()) {
            TreeNode node = q.remove();
            if (node == null) {
                if (curr > max) {
                    max = curr;
                    ans = level;
                }
                curr = 0;
                level++;
                if (!q.isEmpty()) {
                    q.add(null);
                }
            } else {
                curr += node.val;
                if (node.left != null) q.add(node.left);
                if (node.right != null) q.add(node.right);
            }
        }
        return ans;
    }
}
