class Solution {
    public int minOperations(int[] nums, int x) {
        int sum=0,n=nums.length;
        for(int num:nums)sum+=num;
        if(sum==x)return n;
        int curr=0,req=sum-x,i=0,j=0,len=0;
        if(req<0)return -1;
        
        while(j<n){
            curr+=nums[j];
            while(i<j&&curr>req){
                curr-=nums[i];
                i++;
            }
            if(req==curr)len=Math.max(len,j-i+1);
            j++;
        }
        return len==0?-1:n-len;
    }
}
