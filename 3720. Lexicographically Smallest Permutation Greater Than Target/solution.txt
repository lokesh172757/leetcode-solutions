class Solution {
    static class Pair {
        String ans;
        boolean isGreater;

        public Pair(String s, boolean isGreater) {
            this.isGreater = isGreater;
            this.ans = s;
        }
    }

    public static Pair helper(StringBuilder s, String t, int i, int arr[]) {
        if (i >= t.length()) {
            String ans = s.toString();

            if (ans.compareTo(t) > 0) {
                return new Pair(ans, true);
            }

            return new Pair("", false);
        }

        int idx = (int)(t.charAt(i) - 'a');

        if (arr[idx] != 0) {
            s.append((char)(idx + 'a'));
            arr[idx]--;

            Pair p = helper(s, t, i + 1, arr);

            arr[idx]++;
            s.setLength(s.length() - 1);
            if (p.isGreater) {
                return p;
            }
        }

        for (int j = idx + 1; j < 26; j++) {
            if (arr[j] != 0) {

                s.append((char)(j + 'a'));
                arr[j]--;
                StringBuilder ans = new StringBuilder(s);

                for (int k = 0; k < 26; k++) {
                    while (arr[k] > 0) {
                        ans.append((char)(k + 'a'));
                        arr[k]--;
                    }
                }
                for (int k = 0; k < 26; k++) {
                    arr[k] = 0;
                }
                return new Pair(ans.toString(), true);
            }
        }

        return new Pair("", false);
    }

    public String lexGreaterPermutation(String s, String target) {
        int arr[] = new int[26];

        for (int i = 0; i < s.length(); i++) {
            arr[(int)(s.charAt(i) - 'a')]++;
        }

        Pair p = helper(new StringBuilder(""), target, 0, arr);

        return p.ans;
    }
}
