class Solution {
    public static boolean game(char ch,int n,int arr[]){
        if(n<=1){
            return ch=='b';
        }
        if(arr[n]!=0){
            if(arr[n]==1){
                return true;
            }
            return false;
        }
        for(int i=1;i<n;i++){
            if(n%i==0){
                if(ch=='a'){
                    if(game('b',n-i,arr)==true){
                        arr[n]=1;
                        return true;
                    }
                }
                else {
                    if(game('a',n-i,arr)==false){
                        arr[n]=2;
                        return false;
                    }
                }
            }
        }
        if(ch=='b'){
            arr[n]=1;
            return true;
        }
        arr[n]=2;
        return false;

    }

    public boolean divisorGame(int n) {
        int arr[]=new int[n+1];
        return game('a',n,arr);
    }
}
