class Solution {
    public int numJewelsInStones(String jewels, String stones) {
        int ans=0;
        int arr[]=new int[52];
        for(int i=0;i<jewels.length();i++){
            char ch=jewels.charAt(i);
            if(ch>='a'){
                int idx=(int)(ch-'a');
                arr[26+idx]++;
            }
            else{
                int idx=(int)(ch-'A');
                arr[idx]++;
            }
        }
        for(int i=0;i<stones.length();i++){
            char ch=stones.charAt(i);
            if(ch>='a'){
                int idx=(int)(ch-'a');
                if(arr[26+idx]!=0){
                    ans++;
                }
            }
            else{
                int idx=(int)(ch-'A');
                if(arr[idx]!=0){
                    ans++;
                }
            }
        }
        return ans;
    }
}
