class Solution {
    public int countPartitions(int[] nums) {
        int sum = 0;
        for (int x : nums) sum += x;

        int count = 0, prefix = 0;

        for (int i = 0; i < nums.length - 1; i++) {
            prefix += nums[i];
            if ((sum - 2 * prefix) % 2 == 0) {
                count++;
            }
        }
        return count;
    }
}
