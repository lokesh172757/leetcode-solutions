class Solution {
    public void wiggleSort(int[] nums) {
        int n=nums.length;
        int arr[]=new int[n];
        for(int i=0;i<n;i++){
            arr[i]=nums[i];
        }
        Arrays.sort(arr);
        int i=n-1;
        for(int j=1;j<n;j+=2){
            nums[j]=arr[i];
            i--;
        }
        for(int j=0;j<n;j+=2){
            nums[j]=arr[i];
            i--;
        }
    }
}
