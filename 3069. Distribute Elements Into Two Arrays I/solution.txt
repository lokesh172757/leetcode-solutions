class Solution {
    public int[] resultArray(int[] nums) {
        int n=nums.length,i=0,j=1;
        int ans[]=new int[n];
        ans[0]=nums[0];
        nums[0]=0;
        for(int k=2;k<n;k++){
            if(ans[i]>nums[j]){
                ans[++i]=nums[k];
                nums[k]=0;
            }
            else j=k;
        }
        for(int k=0;k<n;k++){
            if(nums[k]!=0)ans[++i]=nums[k];
        }
        return ans;
    }
}
