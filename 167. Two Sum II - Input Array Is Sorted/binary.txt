class Solution {
    public static int find(int t,int nums[]){
        int s=0,e=nums.length-1;
        while(s<=e){
            int mid=s+(e-s)/2;
            if(nums[mid]==t){
                return mid;
            }
            else if(nums[mid]<t){
                s=mid+1;
            }
            else{
                e=mid-1;
            }
        }
        return -1;
    }
    public int[] twoSum(int[] numbers, int target) {
        // Binary search
        int i=0,j=-1;
        while(i<numbers.length){
            int t=target-numbers[i];
            if(t==numbers[i])return new int[]{i+1,i+2};
            j=find(t,numbers);
            if(j!=-1)break;
            i++;
        }
        return new int[]{i+1,j+1};
    }
}
