class Solution {
    public String reverseByType(String s) {
        int n=s.length(),i=0,j=n-1;
        char arr[]=s.toCharArray();
        while(i<j){
            while(i<j){
                char ch=arr[i];
                if(ch>='a'&&ch<='z')break;
                i++;
            }
            while(i<j){
                char ch=arr[j];
                if(ch>='a'&&ch<='z')break;
                j--;
            }
            char temp=arr[i];
            arr[i]=arr[j];
            arr[j]=temp;
            i++;
            j--;
        }
        i=0;
        j=n-1;
        while(i<j){
            while(i<j){
                char ch=arr[i];
                if(ch<'a'||ch>'z')break;
                i++;
            }
            while(i<j){
                char ch=arr[j];
                if(ch<'a'||ch>'z')break;
                j--;
            }
            char temp=arr[i];
            arr[i]=arr[j];
            arr[j]=temp;
            i++;
            j--;
        }
        return new String(arr);
    }
}
