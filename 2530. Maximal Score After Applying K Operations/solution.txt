class Solution {
    public long maxKelements(int[] nums, int k) {
        PriorityQueue<Integer>pq=new PriorityQueue<>(Collections.reverseOrder());
        for(int i=0;i<nums.length;i++){
            pq.add(nums[i]);
        }
        long ans=0;
        while(!pq.isEmpty()&&k>0){
            int n=pq.remove();
            ans+=n;
            n=(int)Math.ceil(n/3.0);
            if(n>0){
                pq.add(n);
            }
            k--;
        }
        return ans;
    }
}
