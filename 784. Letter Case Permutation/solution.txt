class Solution {
    public static void helper(String s,StringBuilder sb,List<String>ans,int i){
        if(i==s.length()){
            ans.add(sb.toString());
            return;
        }
        char ch=s.charAt(i);
        sb.append(ch);
        helper(s,sb,ans,i+1);
        sb.setLength(sb.length()-1);
        if((ch>='a'&&ch<='z')||(ch>='A'&&ch<='Z')){
            if(ch>='a'&&ch<='z'){
                sb.append((char)(ch-32));
            }
            else{
                sb.append((char)(ch+32));
            }
            helper(s,sb,ans,i+1);
            sb.setLength(sb.length()-1);
        }
        
    }
    public List<String> letterCasePermutation(String s) {
        List<String>ans=new ArrayList<>();
        helper(s,new StringBuilder(""),ans,0);
        return ans;
    }
}
