class Solution {

    public static boolean check(int[][] grid, int i, int j, int k) {
        int sum = 0;

        for (int r = i; r < i + k; r++) {
            int currSum = 0;
            for (int c = j; c < j + k; c++) {
                currSum += grid[r][c];
            }
            if (r == i) sum = currSum;
            else if (currSum != sum) return false;
        }

        for (int c = j; c < j + k; c++) {
            int currSum = 0;
            for (int r = i; r < i + k; r++) {
                currSum += grid[r][c];
            }
            if (currSum != sum) return false;
        }

        int diag1 = 0;
        for (int l = 0; l < k; l++) {
            diag1 += grid[i + l][j + l];
        }
        if (diag1 != sum) return false;

        int diag2 = 0;
        for (int l = 0; l < k; l++) {
            diag2 += grid[i + l][j + k - 1 - l];
        }
        if (diag2 != sum) return false;

        return true;
    }

    public int largestMagicSquare(int[][] grid) {
        int m = grid.length, n = grid[0].length;
        int ans = 1;

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 2; i + k <= m && j + k <= n; k++) {
                    if (check(grid, i, j, k)) {
                        ans = Math.max(ans, k);
                    }
                }
            }
        }
        return ans;
    }
}
