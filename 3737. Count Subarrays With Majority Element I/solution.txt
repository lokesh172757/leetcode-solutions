class Solution {
    public int countMajoritySubarrays(int[] nums, int target) {
        int n=nums.length;
        int pre[]=new int[n];
        pre[0]=nums[0]==target?1:0;
        for(int i=1;i<n;i++){
            if(nums[i]==target)pre[i]++;
            pre[i]+=pre[i-1];
        }
        int ans=0;
        for(int i=0;i<n;i++){
            for(int j=i;j<n;j++){
                int freq=i==0?pre[j]:pre[j]-pre[i-1];
                if(freq>(j-i+1)/2)ans++;
            }
        }
        return ans;
    }
}
