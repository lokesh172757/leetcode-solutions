class Solution {
    static int time;
    public static void dfs(ArrayList<Integer>graph[],int disc[],int low[],boolean vis[],int curr,int par,int time,List<List<Integer>>ans){
        vis[curr]=true;
        disc[curr]=low[curr]=++time;
        for(int i=0;i<graph[curr].size();i++){
            int dest=graph[curr].get(i);
            if(dest==par)continue;
            else if(!vis[dest]){
                dfs(graph,disc,low,vis,dest,curr,time,ans);
                low[curr]=Math.min(low[curr],low[dest]);
                if(disc[curr]<low[dest]){
                    ArrayList<Integer>l=new ArrayList<>();
                    l.add(curr);
                    l.add(dest);
                    ans.add(l);
                }
            }
            else{
                low[curr]=Math.min(low[curr],disc[dest]);
            }
        }
    }
    public List<List<Integer>> criticalConnections(int n, List<List<Integer>> connections) {
        time=0;
        int disc[]=new int[n],low[]=new int[n];
        boolean vis[]=new boolean[n];
        List<List<Integer>>ans=new ArrayList<>();
        ArrayList<Integer>graph[]=new ArrayList[n];
        for(int i=0;i<n;i++){
            graph[i]=new ArrayList<>();
        }
        for(int i=0;i<connections.size();i++){
            graph[connections.get(i).get(0)].add(connections.get(i).get(1));
            graph[connections.get(i).get(1)].add(connections.get(i).get(0));
        }
        for(int i=0;i<n;i++){
            if(!vis[i]){
                dfs(graph,disc,low,vis,i,-1,time,ans);
            }
        }
        return ans;
    }
}
