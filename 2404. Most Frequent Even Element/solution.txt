class Solution {
    public int mostFrequentEven(int[] nums) {
        int max=Integer.MIN_VALUE;
        for(int n:nums){
            if(n>max)max=n;
        }
        int fr[]=new int[max+1];
        for(int n:nums){
            fr[n]++;
        }
        int f=0,ans=-1;
        for(int i=0;i<=max;i+=2){
            if(fr[i]!=0&&fr[i]>f){
                f=fr[i];
                ans=i;
            }
        }
        return ans;
    }
}
