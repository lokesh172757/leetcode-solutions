class Solution {
    public static int largeRectHist(int arr[]){
        int n=arr.length;
        int leftmin[]=new int[n],rightmin[]=new int[n];
        Stack<Integer>st=new Stack<>();
        for(int i=0;i<n;i++){
            while(!st.isEmpty()&&arr[i]<=arr[st.peek()])st.pop();
            if(st.isEmpty())leftmin[i]=-1;
            else leftmin[i]=st.peek();
            st.push(i);
        }
        st.clear();
        for(int i=n-1;i>=0;i--){
            while(!st.isEmpty()&&arr[i]<=arr[st.peek()])st.pop();
            if(st.isEmpty())rightmin[i]=n;
            else rightmin[i]=st.peek();
            st.push(i);
        }
        int max=0;
        for(int i=0;i<n;i++){
            max=Math.max(max,(rightmin[i]-leftmin[i]-1)*arr[i]);
        }
        return max;
    }
    public int maximalRectangle(char[][] matrix) {
        int ans=0,m=matrix.length,n=matrix[0].length;
        int arr[][]=new int[m][n];
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                arr[i][j]=(int)(matrix[i][j]-'0');
                if((i!=0)&&(arr[i][j]!=0))arr[i][j]+=arr[i-1][j];
            }
            
            ans=Math.max(ans,largeRectHist(arr[i]));
        }
        return ans;
    }
}
