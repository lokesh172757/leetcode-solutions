class Solution {
    public int mostBooked(int n, int[][] meetings) {

        // Sort meetings by start time
        Arrays.sort(meetings, (a, b) -> a[0] - b[0]);

        // Min heap of free rooms (by room index)
        PriorityQueue<Integer> freeRooms = new PriorityQueue<>();

        // Min heap of busy rooms: [endTime, roomIndex]
        PriorityQueue<long[]> busyRooms = new PriorityQueue<>(
            (a, b) -> a[0] == b[0] ? Long.compare(a[1], b[1]) : Long.compare(a[0], b[0])
        );

        // Initially all rooms are free
        for (int i = 0; i < n; i++) {
            freeRooms.offer(i);
        }

        int[] freq = new int[n];

        for (int[] meeting : meetings) {
            long start = meeting[0];
            long end = meeting[1];
            long duration = end - start;

            // Free rooms that have completed before current meeting
            while (!busyRooms.isEmpty() && busyRooms.peek()[0] <= start) {
                freeRooms.offer((int) busyRooms.poll()[1]);
            }

            if (!freeRooms.isEmpty()) {
                // Assign meeting to free room
                int room = freeRooms.poll();
                busyRooms.offer(new long[]{end, room});
                freq[room]++;
            } else {
                // Delay meeting
                long[] earliest = busyRooms.poll();
                long newEnd = earliest[0] + duration;
                int room = (int) earliest[1];
                busyRooms.offer(new long[]{newEnd, room});
                freq[room]++;
            }
        }

        // Find room with maximum frequency
        int ans = 0;
        for (int i = 1; i < n; i++) {
            if (freq[i] > freq[ans]) {
                ans = i;
            }
        }

        return ans;
    }
}
