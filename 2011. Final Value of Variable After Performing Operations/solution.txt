class Solution {
    public int finalValueAfterOperations(String[] operations) {
        int x=0;
        for(int i=0;i<operations.length;i++){
            String str=operations[i];
            if(str.charAt(0)=='-'||str.charAt(str.length()-1)=='-'){
                x--;
            }
            else{
                x++;
            }
        }
        return x;
    }
}
