class Solution {
    public String smallestPalindrome(String s) {
        int arr[]=new int[26];
        for(int i=0;i<s.length();i++){
            arr[(int)(s.charAt(i)-'a')]++;
        }
        char last='%';
        StringBuilder start=new StringBuilder(""),end=new StringBuilder("");
        for(int i=0;i<26;i++){
            char c=(char)('a'+i);
            if(arr[i]%2==1)last=c;
            arr[i]--;
            while(arr[i]>0){
                start.append(c);
                end.append(c);
                arr[i]-=2;
            }
        }
        if(last!='%')start.append(last);
        start.append(end.reverse());
        return start.toString();
    }
}
