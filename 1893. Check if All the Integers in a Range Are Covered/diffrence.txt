class Solution {
    public boolean isCovered(int[][] ranges, int left, int right) {
        //diffrence array approach
        int arr[]=new int[right+1];
        for(int ar[]:ranges){
            if(ar[0]>right)continue;
            arr[ar[0]]++;
            if(ar[1]<right){
                arr[ar[1]+1]--;
            }
        }
        for(int i=1;i<=right;i++){
            arr[i]+=arr[i-1];
        }
        for(int i=left;i<=right;i++){
            if(arr[i]==0)return false;
        }
        return true;
    }
}
