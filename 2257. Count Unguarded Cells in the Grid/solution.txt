class Solution {
    public int countUnguarded(int m, int n, int[][] guard, int[][] walls) {
        int arr[][] = new int[m][n];
        
        // mark walls
        for (int i = 0; i < walls.length; i++) {
            arr[walls[i][0]][walls[i][1]] = 2;
        }
        // mark guards
        for (int i = 0; i < guard.length; i++) {
            arr[guard[i][0]][guard[i][1]] = 3;
        }
        
        int guarded = 0;
        for (int i = 0; i < guard.length; i++) {
            int row = guard[i][0], col = guard[i][1];
            int r, c;
            
            // right
            r = row; c = col + 1;
            while (c < n) {
                if (arr[r][c] == 2 || arr[r][c] == 3) break;
                if (arr[r][c] != 1) {
                    arr[r][c] = 1;
                    guarded++;
                }
                c++;
            }
            
            // left
            r = row; c = col - 1;
            while (c >= 0) {
                if (arr[r][c] == 2 || arr[r][c] == 3) break;
                if (arr[r][c] != 1) {
                    arr[r][c] = 1;
                    guarded++;
                }
                c--;
            }
            
            // up
            r = row - 1; c = col;
            while (r >= 0) {
                if (arr[r][c] == 2 || arr[r][c] == 3) break;
                if (arr[r][c] != 1) {
                    arr[r][c] = 1;
                    guarded++;
                }
                r--;
            }
            
            // down
            r = row + 1; c = col;
            while (r < m) {
                if (arr[r][c] == 2 || arr[r][c] == 3) break;
                if (arr[r][c] != 1) {
                    arr[r][c] = 1;
                    guarded++;
                }
                r++;
            }
        }

        return m * n - guarded - walls.length - guard.length;
    }
}
