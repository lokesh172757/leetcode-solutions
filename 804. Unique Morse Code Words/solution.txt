class Solution {
    public int uniqueMorseRepresentations(String[] words) {
        String arr[]={".-","-...","-.-.","-..",".","..-.","--.","....","..",".---","-.-",".-..","--","-.","---",".--.","--.-",".-.","...","-","..-","...-",".--","-..-","-.--","--.."};
        HashSet<String>set=new HashSet<>();
        for(String str:words){
            StringBuilder sb=new StringBuilder("");
            for(int i=0;i<str.length();i++){
                char c=str.charAt(i);
                sb.append(arr[(int)(c-'a')]);

            }
            set.add(sb.toString());
        }
        return set.size();
    }
}
