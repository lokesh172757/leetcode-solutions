class Solution {
    public static int minXor(int[][] grid, Integer[][][] dp, int i, int j, int xor) {
        int m = grid.length, n = grid[0].length;

        if (i >= m || j >= n) return Integer.MAX_VALUE;

        xor ^= grid[i][j];

        if (i == m - 1 && j == n - 1) return xor;

        if (dp[i][j][xor] != null) return dp[i][j][xor];

        int ans1 = minXor(grid, dp, i + 1, j, xor);
        int ans2 = minXor(grid, dp, i, j + 1, xor);

        return dp[i][j][xor] = Math.min(ans1, ans2);
    }

    public int minCost(int[][] grid) {
        int m = grid.length, n = grid[0].length;
        Integer[][][] dp = new Integer[m][n][1024];

        return minXor(grid, dp, 0, 0, 0);
    }
}
