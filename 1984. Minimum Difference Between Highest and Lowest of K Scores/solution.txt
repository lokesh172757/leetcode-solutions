class Solution {
    public int minimumDifference(int[] nums, int k) {
        if(k==1)return 0;
        Arrays.sort(nums);
        int i=0,j=k-1,ans=Integer.MAX_VALUE,n=nums.length;
        while(j<n){
            ans=Math.min(ans,nums[j]-nums[i]);
            j++;i++;
        }
        return ans;

    }
}
