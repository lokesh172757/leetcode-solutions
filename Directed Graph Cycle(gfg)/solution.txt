class Solution {
    public static boolean isCyclicUtil(ArrayList<Integer>graph[],boolean vis[],boolean st[],int src){
        vis[src]=true;
        st[src]=true;
        for(int i=0;i<graph[src].size();i++){
            int n=graph[src].get(i);
            if(st[n]){
                return true;
            }
            else if(!vis[n]&&isCyclicUtil(graph,vis,st,n)){
                return true;
            }
        }
        st[src]=false;
        return false;
    }
    public boolean isCyclic(int V, int[][] edges) {
        ArrayList<Integer>graph[]=new ArrayList[V];
        for(int i=0;i<V;i++){
            graph[i]=new ArrayList<>();
            
        }
        for(int i=0;i<edges.length;i++){
            graph[edges[i][0]].add(edges[i][1]);
        }
        boolean st[]=new boolean[V];
        boolean vis[]=new boolean[V];
        for(int i=0;i<V;i++){
            if(!vis[i]){
                if(isCyclicUtil(graph,vis,st,i)){
                    return true;
                }
            }
        }
        return false;
        
    }
}
