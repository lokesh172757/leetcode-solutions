class MyCalendar {
    TreeMap<Integer,Integer>map;
    public MyCalendar() {
        map=new TreeMap<>();
    }
    
    public boolean book(int startTime, int endTime) {
        map.put(startTime,map.getOrDefault(startTime,0)+1);
        map.put(endTime,map.getOrDefault(endTime,0)-1);
        int curr=0;
        for(Integer num:map.values()){
            curr+=num;
            if(curr>1){
                map.put(endTime,map.getOrDefault(endTime,0)+1);
                map.put(startTime,map.getOrDefault(startTime,0)-1);
                return false;
            }
        }
        return true;
    }
}
