class Solution {
    public int[] leftRightDifference(int[] nums) {
        int n=nums.length,sum=0;
        for(int i=0;i<n;i++){
            sum+=nums[i];
        }
        int leftSum=0;
        int ans[]=new int[n];
        for(int i=0;i<n;i++){
            ans[i]=Math.abs(2*leftSum-sum+nums[i]);
            leftSum+=nums[i];
        }
        return ans;
    }
}
