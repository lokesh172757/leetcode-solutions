// optimized using 1D dp 
class Solution {
    public int longestPalindromeSubseq(String s) {
        int n=s.length();
        int dp[]=new int[n+1];
        for(int i=1;i<=n;i++){
            int prevDiag=0;
            for(int j=1;j<=n;j++){
                int temp=dp[j];
                if(s.charAt(i-1)==s.charAt(n-j)){
                    dp[j]=prevDiag+1;
                }
                else{
                    dp[j]=Math.max(dp[j],dp[j-1]);
                }
                prevDiag=temp;
            }
        }
        return dp[n];
    }
}
