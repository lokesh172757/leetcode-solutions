class Solution {
    public int minOperations(int[] nums) {
        Stack<Integer>st=new Stack<>();
        int ans=0;
        for(int num:nums){
            if(num==0){
                st.clear();
                continue;
            }
            while(!st.isEmpty()&&st.peek()>num){
                st.pop();
            }
            if(st.isEmpty()||st.peek()<num){
                st.push(num);
                ans++;
            }
        }
        return ans;
    }
}
