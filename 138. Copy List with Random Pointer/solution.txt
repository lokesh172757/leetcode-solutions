class Solution {
    public static Node createList(Node head,HashMap<Node,Node>map){
        Node ans=null,temp=null;
        
        while(head!=null){
            Node nn=new Node(head.val);
            if(ans==null){
                ans=nn;
                temp=nn;
            }
            else{
                temp.next=nn;
                temp=nn;
            }
            map.put(head,nn);
            head=head.next;
        }
        return ans;
    }
    public Node copyRandomList(Node head) {
        HashMap<Node,Node>map=new HashMap<>();
        map.put(null,null);
        Node ans=createList(head,map);
        Node temp1=head,temp2=ans;
        while(temp1!=null){
            temp2.random=map.get(temp1.random);
            temp1=temp1.next;
            temp2=temp2.next;
        }
        return ans;
    }
}
