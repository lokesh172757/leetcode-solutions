class Solution {
    public int countDigitOccurrences(int[] nums, int digit) {
        int arr[]=new int[10];
        for(int i=0;i<nums.length;i++){
            int n=nums[i];
            if(n==0){
                arr[0]++;
                continue;
            }
            while(n>0){
                arr[n%10]++;
                n/=10;
            }
        }
        return arr[digit];
    }
}
