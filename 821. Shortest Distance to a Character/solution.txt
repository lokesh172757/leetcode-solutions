class Solution {
    public int[] shortestToChar(String s, char c) {
        int n=s.length();
        int ans[]=new int[n];
        HashSet<Integer>set=new HashSet<>();
        for(int i=0;i<n;i++){
            if(s.charAt(i)==c)set.add(i);
        }
        if(set.size()==n)return ans;
        for(int i=0;i<n;i++){
            int min=n;
            for(Integer j:set){
                min=Math.min(min,Math.abs(i-j));
            }
            ans[i]=min;
        }
        return ans;
    }
}
