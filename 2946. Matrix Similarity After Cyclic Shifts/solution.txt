class Solution {
    public boolean areSimilar(int[][] mat, int k) {
        int m=mat.length,n=mat[0].length;
        k%=n;
        if(k==0)return true;
        for(int i=0;i<m;i++){
            int st=i%2==0?k:n-k;
            for(int j=0;j<n;j++){
                if(mat[i][j]!=mat[i][st%n])return false;
                st++;
            }
        }
        return true;
    }
}
