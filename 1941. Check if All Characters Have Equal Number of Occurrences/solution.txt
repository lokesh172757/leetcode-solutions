class Solution {
    public boolean areOccurrencesEqual(String s) {
        int freq[]=new int[26];
        for(int i=0;i<s.length();i++){
            freq[s.charAt(i)-'a']++;
        }
        int count=-1;
        for(int i=0;i<26;i++){
            if(freq[i]!=0){
                if(count==-1)count=freq[i];
                else if(freq[i]!=count)return false; 
            }
        }
        return true;
    }
}
