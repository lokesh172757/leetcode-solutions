class Solution {
    public int numberOfSubmatrices(char[][] grid) {
        int m=grid.length,n=grid[0].length;
        int y[][]=new int[m][n];
        int x[][]=new int[m][n];
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]=='Y')y[i][j]=1;
                if(grid[i][j]=='X')x[i][j]=1;
                if(j>0){
                    y[i][j]+=y[i][j-1];
                    x[i][j]+=x[i][j-1];
                }
            }
        }
        int ans=0;
        for(int j=0;j<n;j++){
            int xsum=0,ysum=0;
            for(int i=0;i<m;i++){
                xsum+=x[i][j];
                ysum+=y[i][j];
                if(xsum!=0&&xsum==ysum)ans++;
            }
        }
        return ans;
    }
}
