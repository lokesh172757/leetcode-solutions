class Solution {
    public int totalSteps(int[] nums) {
        int n=nums.length;
        int dp[]=new int[n];
        Deque<Integer>st=new ArrayDeque<>();
        for(int i=0;i<n;i++){
            int max=0;
            while(!st.isEmpty()&&nums[st.peek()]<=nums[i]){
                max=Math.max(dp[st.peek()],max);
                st.pop();
            }
            if(st.isEmpty())dp[i]=0;
            else dp[i]=max+1;
            st.push(i);
        }
        int max=0;
        for(int i=0;i<n;i++){
            max=Math.max(max,dp[i]);
        }
        return max;
    }
}
