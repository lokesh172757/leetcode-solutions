class Solution {
    public int minOperations(List<Integer> nums, int k) {
        int n=nums.size();
        if(k==n)return n;
        int remaining=k;
        boolean picked[]=new boolean[k+1];
        for(int i=n-1;i>=0;i--){
            int num=nums.get(i);
            if(num<=k&&!picked[num]){
                picked[num]=true;
                if(--remaining==0)return n-i;
            }
        }
        return 0;
    }
}
