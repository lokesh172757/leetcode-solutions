class Solution {
    public List<String> removeAnagrams(String[] words) {
        List<String>ans=new ArrayList<>();
        String ar[]=new String[words.length];
        for(int i=0;i<words.length;i++){
            char arr[]=words[i].toCharArray();
            Arrays.sort(arr);
            ar[i]=new String(arr);
        }
        int i=0,j=1;
        while(j<words.length){
            if(!(ar[i].equals(ar[j]))){
                ans.add(words[i]);
                i=j;
            }
            j++;
        }
        ans.add(words[i]);
        return ans;

    }
}
