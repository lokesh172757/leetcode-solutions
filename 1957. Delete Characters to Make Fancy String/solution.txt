class Solution {
    public String makeFancyString(String s) {
        StringBuilder sb=new StringBuilder("");
        int cnt=1;
        char prev=s.charAt(0);
        sb.append(prev);
        for(int i=1;i<s.length();i++){
            char ch=s.charAt(i);
            if(ch==prev){
                cnt++;
                if(cnt<3){
                    sb.append(ch);
                }
            }
            else{
                sb.append(ch);
                prev=ch;
                cnt=1;
            }
            
        }
        return sb.toString();
    }
}
