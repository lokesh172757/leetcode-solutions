class Solution {
    public boolean areNumbersAscending(String s) {
        String arr[]=s.split(" ");
        int prev=-1;
        for(int i=0;i<arr.length;i++){
            if(arr[i].charAt(0)>='0'&&arr[i].charAt(0)<='9'){
                int num=Integer.parseInt(arr[i]);
                if(prev>=num)return false;
                else{
                    prev=num;
                }
            }
        }
        return true;
    }
}
