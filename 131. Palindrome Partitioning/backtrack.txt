class Solution {
    static List<List<String>>ans;
    public static boolean isPal(String s){
        int i=0,j=s.length()-1;
        while(i<j){
            if(s.charAt(i)!=s.charAt(j))return false;
            i++;
            j--;
        }
        return true;
    }
    public static void helper(String s,ArrayList<String>curr){
        if(s.length()==0){
            if(curr.size()!=0)ans.add(new ArrayList(curr));
            return;
        }
        for(int i=0;i<s.length();i++){
            String f=s.substring(0,i+1);
            if(isPal(f)){
                curr.add(f);
                helper(s.substring(i+1),curr);
                curr.remove(curr.size()-1);
            }
        }
    }
    public List<List<String>> partition(String s) {
        ans=new ArrayList<>();
        if(s=="")return ans;
        helper(s,new ArrayList<>());
        return ans;
    }
}
