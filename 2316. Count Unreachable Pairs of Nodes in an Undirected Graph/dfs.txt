class Solution {
    public static long dfs(ArrayList<Integer>graph[]){
        int n=graph.length;
        
        long ans=0,p=0;
        boolean vis[]=new boolean[n];
        for(int i=0;i<n;i++){
            if(!vis[i]){
                long x=dfsUtil(graph,vis,i);
                ans+=p*x;
                p+=x;
            }
        }
        return ans;
    }
    public static long dfsUtil(ArrayList<Integer>graph[],boolean vis[],int src){
        long ans=1;
        vis[src]=true;
        for(int i=0;i<graph[src].size();i++){
            int next=graph[src].get(i);
            if(!vis[next]){
                ans+=dfsUtil(graph,vis,next);
            }
        }
        return ans;
    }
    public long countPairs(int n, int[][] edges) {
        ArrayList<Integer>graph[]=new ArrayList[n];
        for(int i=0;i<n;i++){
            graph[i]=new ArrayList<>();
        }
        for(int i=0;i<edges.length;i++){
            int a=edges[i][0],b=edges[i][1];
            graph[a].add(b);
            graph[b].add(a);
        }
        return dfs(graph);
    }
}
