class Solution {
    public boolean sumGame(String num) {
        int leftSum=0,rightSum=0,leftq=0,rightq=0,n=num.length();

        for(int i=0;i<n/2;i++){
            char ch=num.charAt(i);
            if(ch=='?') leftq++;
            else leftSum+=(int)(ch-'0');
        }

        for(int i=n/2;i<n;i++){
            char ch=num.charAt(i);
            if(ch=='?') rightq++;
            else rightSum+=(int)(ch-'0');
        }

        if((leftq+rightq)%2==1) return true;

        int diff=leftSum-rightSum;

        return diff != 9*(rightq-leftq)/2;
    }
}
