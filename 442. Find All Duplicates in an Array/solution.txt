class Solution {
    public List<Integer> findDuplicates(int[] nums) {
        List<Integer>ans=new ArrayList<>();
        int n=nums.length;
        int arr[]=new int[n+1];
        for(int i=0;i<n;i++){
            if(arr[nums[i]]==1){
                ans.add(nums[i]);
            }
            else{
                arr[nums[i]]++;
            }
        }
        return ans;
    }
}
