class Solution {
    public static ListNode rev(ListNode curr, ListNode end){
        ListNode prev = null, next;
        while (curr != end) {
            next = curr.next;
            curr.next = prev;
            prev = curr;
            curr = next;
        }
        curr.next = prev;
        return curr;
    }

    public ListNode reverseKGroup(ListNode head, int k) {
        if (head == null || head.next == null) return head;

        ListNode d = new ListNode(-1);
        d.next = head;

        ListNode f = head, h = d;
        int count = 0;

        while (head != null) {
            count++;
            if (count == k) {
                ListNode l = head.next;
                h.next = rev(f, head);
                f.next = l;
                h = f;
                f = l;
                head = l;
                count = 0;
            } else {
                head = head.next;
            }
        }
        return d.next;
    }
}
