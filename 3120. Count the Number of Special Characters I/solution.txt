class Solution {
    public int numberOfSpecialChars(String word) {
        int arr[]=new int[52];
        for(int i=0;i<word.length();i++){
            char ch=word.charAt(i);
            if(ch>'Z'){
                int idx=(int)(ch-'a');
                arr[idx+26]++;
            }
            else{
                int idx=(int)(ch-'A');
                arr[idx]++;
            }
        }
        int ans=0;
        for(int i=0;i<26;i++){
            if(arr[i]!=0&&arr[i+26]!=0){
                ans++;
            }
        }
        return ans;
        
    }
}
