class Solution {
    public int minFlips(String s) {
        int m = s.length(),n=2*m;

        int arr[] = new int[n];

        // Pattern 010101...
        for(int i = 0; i < n; i++){
            if(i % 2 == 0 && s.charAt(i%m) != '0') arr[i]++;
            else if(i % 2 == 1 && s.charAt(i%m) != '1') arr[i]++;
        }

        for(int i = 1; i < n; i++){
            arr[i] += arr[i-1];
        }

        int ans = Integer.MAX_VALUE;

        for(int i = 0; i + m - 1 < n; i++){
            int flips = arr[i + m - 1] - (i > 0 ? arr[i-1] : 0);
            ans = Math.min(ans, flips);
        }

        // Pattern 101010...
        arr = new int[n];

        for(int i = 0; i < n; i++){
            if(i % 2 == 0 && s.charAt(i%m) != '1') arr[i]++;
            else if(i % 2 == 1 && s.charAt(i%m) != '0') arr[i]++;
        }

        for(int i = 1; i < n; i++){
            arr[i] += arr[i-1];
        }

        for(int i = 0; i + m - 1 < n; i++){
            int flips = arr[i + m - 1] - (i > 0 ? arr[i-1] : 0);
            ans = Math.min(ans, flips);
        }

        return ans;
    }
}
