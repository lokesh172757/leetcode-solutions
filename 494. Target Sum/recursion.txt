class Solution {
    public static int helper(int nums[],int t,int idx){
        if(idx==nums.length){
            if(t==0){
                return 1;
            }
            return 0;
        }
        return helper(nums,t+nums[idx],idx+1)+helper(nums,t-nums[idx],idx+1);
    }
    public int findTargetSumWays(int[] nums, int target) {
        
        return helper(nums,target,0);
    }
}
