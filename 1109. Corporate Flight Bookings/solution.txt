class Solution {
    public int[] corpFlightBookings(int[][] bookings, int n) {
        int ans[]=new int[n];
        for(int i=0;i<bookings.length;i++){
            int a=bookings[i][0]-1,b=bookings[i][1],c=bookings[i][2];
            ans[a]+=c;
            if(b<n)ans[b]-=c;
        }
        for(int i=1;i<n;i++){
            ans[i]+=ans[i-1];
        }
        return ans;
    }
}
