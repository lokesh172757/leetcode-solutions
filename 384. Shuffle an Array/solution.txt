class Solution {
    int nums[];
    public Solution(int[] nums) {
        this.nums=nums;
    }
    
    public int[] reset() {
        return nums;
    }
    
    public int[] shuffle() {
        int n=nums.length;
        int arr[]=new int[n];
        for(int i=0;i<n;i++){
            int idx= ThreadLocalRandom.current().nextInt(0, n);
            while(arr[idx]!=0)idx=ThreadLocalRandom.current().nextInt(0, n);
            arr[idx]=nums[i];
        }
        return arr;
    }
}
