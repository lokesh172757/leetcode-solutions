
class Solution {
    public static int helper(TreeNode root,int n){
        if(root==null){
            return n;
        }
        if(root.left==null&&root.right==null){
            root.val+=n;
            return root.val;
        }
        
        int r=helper(root.right,n);
        root.val+=r;
        return helper(root.left,root.val);
        
    }
    public TreeNode convertBST(TreeNode root) {
        if(root==null){
        return root;
        }
        helper(root,0);
        return root;
    }
}
