class Solution {
    public static void helper(List<List<Integer>> rooms,boolean vis[],int src){
        vis[src]=true;
        for(int i=0;i<rooms.get(src).size();i++){
            int dest=rooms.get(src).get(i);
            if(!vis[dest]){
                helper(rooms,vis,dest);
            }
        }

    }
    public boolean canVisitAllRooms(List<List<Integer>> rooms) {
        int n=rooms.size();
        boolean vis[]=new boolean[n];
        helper(rooms,vis,0);
        for(int i=0;i<n;i++){
            if(!vis[i]){
                return false;
            }
        }
        return true;
    }
}
