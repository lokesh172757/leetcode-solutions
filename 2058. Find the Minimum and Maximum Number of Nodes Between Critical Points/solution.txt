class Solution {
    public int[] nodesBetweenCriticalPoints(ListNode head) {
        int ans[]={-1,-1};
        int last=-1,first=-1,min=Integer.MAX_VALUE,i=2;
        ListNode temp=head.next,lastNode=head;
        while(temp.next!=null){
            if((temp.val<temp.next.val&&temp.val<lastNode.val)||(temp.val>temp.next.val&&temp.val>lastNode.val)){
                if(last==-1){
                    first=i;
                    last=i;
                }
                else{
                    min=Math.min(i-last,min);
                    last=i;
                }
            }
            lastNode=temp;
            temp=temp.next;
            i++;
        }
        if(first==last)return ans;
        ans[0]=min;
        ans[1]=last-first;
        return ans;
    }
}
