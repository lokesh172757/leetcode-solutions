class Solution {
    public String digitSum(String s, int k) {
        int n=s.length(),i=0;
        while(n>k){
            StringBuilder sb=new StringBuilder("");
            while(i<n){
                int sum=0;
                for(int j=0;j<k&&i<n;j++){
                    sum+=(int)(s.charAt(i)-'0');
                    i++;
                }
                sb.append(sum+"");
            }
            s=sb.toString();
            n=s.length();
            i=0;
        }
        return s;
    }
}
