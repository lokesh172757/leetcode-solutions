class Solution {
    static class Pair implements Comparable<Pair>{
        int v,r;
        public Pair(int v,int r){
            this.v=v;
            this.r=r;
        }
        @Override
        public int compareTo(Pair p2){
            if(this.r==p2.r){
                return this.v-p2.v;
            }
            return this.r-p2.r;
        }
    }
    public int[] sortByReflection(int[] nums) {
        PriorityQueue<Pair>pq=new PriorityQueue<>();
        for(int i=0;i<nums.length;i++){
            int num=nums[i],rev=0;
            while(num>0){
                rev*=2;
                rev+=num%2;
                num/=2;
            }
            pq.add(new Pair(nums[i],rev));
        }
        int i=0;
        while(!pq.isEmpty()){
            nums[i]=pq.remove().v;
            i++;
        }
        return nums;
    }
}
