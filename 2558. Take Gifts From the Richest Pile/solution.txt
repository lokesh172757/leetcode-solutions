class Solution {
    public long pickGifts(int[] gifts, int k) {
        int n=gifts.length;
        PriorityQueue<Integer>pq=new PriorityQueue<>(Collections.reverseOrder());
        for(int i=0;i<n;i++){
            pq.add(gifts[i]);
        }
        while(k>0){
            pq.add((int)Math.sqrt(pq.remove()));
            k--;
        }
        long ans=0;
        while(!pq.isEmpty())ans+=pq.remove();
        return ans;
    }
}
