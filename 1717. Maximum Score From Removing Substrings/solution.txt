class Solution {
    public int maximumGain(String s, int x, int y) {
        if (x > y) {
            return removePair(s, 'a', 'b', x, y);
        } else {
            return removePair(s, 'b', 'a', y, x);
        }
    }

    private int removePair(String s, char first, char second, int firstGain, int secondGain) {
        StringBuilder sb = new StringBuilder();
        int gain = 0;

        for (char ch : s.toCharArray()) {
            int len = sb.length();
            if (len > 0 && sb.charAt(len - 1) == first && ch == second) {
                sb.deleteCharAt(len - 1);
                gain += firstGain;
            } else {
                sb.append(ch);
            }
        }

        StringBuilder newSb = new StringBuilder();
        for (int i = 0; i < sb.length(); i++) {
            char ch = sb.charAt(i);
            int len = newSb.length();
            if (len > 0 && newSb.charAt(len - 1) == second && ch == first) {
                newSb.deleteCharAt(len - 1);
                gain += secondGain;
            } else {
                newSb.append(ch);
            }
        }

        return gain;
    }
}
