class Solution {
    public boolean kLengthApart(int[] nums, int k) {
        int last=0;
        boolean first=false;
        for(int i=0;i<nums.length;i++){
            if(nums[i]==1){
                if(!first){
                    last=i;
                    first=true;
                }
                else if(i-last-1<k){
                    return false;
                }
                else{
                    last=i;
                }
            }
        }
        return true;
    }
}
