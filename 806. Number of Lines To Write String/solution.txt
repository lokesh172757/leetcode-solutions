class Solution {
    public int[] numberOfLines(int[] widths, String s) {
        int n=s.length(),line=1,width=100;
        for(int i=0;i<n;i++){
            int c=(int)(s.charAt(i)-'a');
            if(width<widths[c]){
                line++;
                width=100;
            }
            width-=widths[c];
        }
        return new int[]{line,100-width};
    }
}
