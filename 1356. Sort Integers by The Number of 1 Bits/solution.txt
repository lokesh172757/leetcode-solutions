class Solution {
    static class Pair implements Comparable<Pair>{
        int val,bit;
        public Pair(int val,int bit){
            this.val=val;
            this.bit=bit;
        }
        @Override
        public int compareTo(Pair p2){
            if(this.bit==p2.bit)return this.val-p2.val;
            return this.bit-p2.bit;
        }
    }
    public int[] sortByBits(int[] arr) {
        PriorityQueue<Integer>twoPow=new PriorityQueue<>();
        PriorityQueue<Pair>pq=new PriorityQueue<>();
        for(int i=0;i<arr.length;i++){
            if((arr[i]&(arr[i]-1))==0)twoPow.add(arr[i]);
            else{
                int num=arr[i],bit=0;
                for(int j=0;j<32&&arr[i]>0;j++){
                    if(arr[i]%2==1)bit++;
                    arr[i]/=2;
                }
                pq.add(new Pair(num,bit));
            }
        }
        int i=0;
        while(!twoPow.isEmpty()){
            arr[i]=twoPow.remove();
            i++;
        }
        while(!pq.isEmpty()){
            arr[i]=pq.remove().val;
            i++;
        }
        return arr;
    }
}
