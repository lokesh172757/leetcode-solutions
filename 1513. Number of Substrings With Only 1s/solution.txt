class Solution {
    public int numSub(String s) {
        int i=0,n=s.length(),mod=(int)1e9+7;
        long ans=0,c=0;
        while(i<n){
            if(s.charAt(i)=='1'){
                c++;
            }
            else if(c>0){
                ans+=((c*(c+1))/2)%mod;
                c=0;
            }
            i++;
        }
        if(c>0){
            ans+=(c*(c+1))/2;
        }
        return (int)ans;
    }
}
