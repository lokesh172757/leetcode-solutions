class Solution {
    public int findPairs(int[] nums, int k) {
        HashSet<Integer> hs=new HashSet<>();
        HashSet<Integer> seen=new HashSet<>();
        int count=0;
        for(int i=0;i<nums.length;i++){
            if(k==0){
                if(hs.contains(nums[i]) && !seen.contains(nums[i])){
                    count++;
                    seen.add(nums[i]);
                }
            }
            hs.add(nums[i]);
        }
        if(k>0){
            for(int n : hs){
                if(hs.contains(n+k)) count++;
            }
        }
        return count;
    }
}
