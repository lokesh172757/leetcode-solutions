class Solution {
    static boolean[] isPrime;
    boolean calculated=false;
    public static void sieve() {
        int n=(int)1e6+1;
        isPrime = new boolean[n];
        Arrays.fill(isPrime, true);
        isPrime[1] = false;
        for (int i = 2; i * i < n; i++) {
            if (isPrime[i]) {
                for (int j = i * i; j < n; j += i) {
                    isPrime[j] = false;
                }
            }
        }
    }
    public static int nextPrime(int n){
        int ans=1;
        for(int i=n+1;i<isPrime.length;i++){
            if(isPrime[i])break;
            ans++;
        }
        return ans;
    }
    public static int nextNotPrime(int n){
        int ans=1;
        for(int i=n+1;i<isPrime.length;i++){
            if(!isPrime[i])break;
            ans++;
        }
        return ans;
    }
    public int minOperations(int[] nums) {
        if(!calculated)sieve();
        int ans=0;
        for(int i=0;i<nums.length;i++){
            int n=nums[i];
            if(i%2==0&&(!isPrime[n])){
                ans+=nextPrime(n);
            }
            else if(i%2==1&&isPrime[n]){
                ans+=nextNotPrime(n);
            }
        }
        return ans;
    }
}
