class Solution {
    public long minimumCost(String source, String target, char[] original, char[] changed, int[] cost) {

        long[][] graph = new long[26][26];

        for (int i = 0; i < 26; i++) {
            for (int j = 0; j < 26; j++) {
                graph[i][j] = Long.MAX_VALUE;
            }
            graph[i][i] = 0;
        }
        for (int i = 0; i < original.length; i++) {
            int u = original[i] - 'a';
            int v = changed[i] - 'a';
            graph[u][v] = Math.min(graph[u][v], cost[i]);
        }
        for (int via = 0; via < 26; via++) {
            for (int i = 0; i < 26; i++) {
                for (int j = 0; j < 26; j++) {
                    if (graph[i][via] != Long.MAX_VALUE && graph[via][j] != Long.MAX_VALUE) {
                        graph[i][j] = Math.min(graph[i][j], graph[i][via] + graph[via][j]);
                    }
                }
            }
        }

        long ans = 0;
        for (int i = 0; i < source.length(); i++) {
            int a = source.charAt(i) - 'a';
            int b = target.charAt(i) - 'a';

            if (a == b) continue;
            if (graph[a][b] == Long.MAX_VALUE) return -1;

            ans += graph[a][b];
        }

        return ans;
    }
}
