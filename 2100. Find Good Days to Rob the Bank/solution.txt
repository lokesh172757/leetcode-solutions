class Solution {
    public List<Integer> goodDaysToRobBank(int[] security, int time) {
        List<Integer>ans=new ArrayList<>();
        int n=security.length;
        int start[]=new int[n];
        int end[]=new int [n];
        int cnt=0;
        for(int i=1;i<n;i++){
            if(security[i]<=security[i-1]){
                cnt++;
            }else{
                cnt=0;
            }
            start[i]=cnt;
        }
        cnt=0;
        for(int i=n-2;i>=0;i--){
            if(security[i]<=security[i+1]){
                cnt++;
            }
            else{
                cnt=0;
            }
            end[i]=cnt;
        }
        for(int i=time;i<n-time;i++){
            if(start[i]>=time&&end[i]>=time){
                ans.add(i);
            }
        }
        return ans;
    }
}
