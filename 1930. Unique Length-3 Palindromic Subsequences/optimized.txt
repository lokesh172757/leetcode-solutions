class Solution {
    public int countPalindromicSubsequence(String s) {
        int n=s.length(),ans=0;
        int first[]=new int[26],last[]=new int[26];
        Arrays.fill(first,-1);
        Arrays.fill(last,-1);
        for(int i=0;i<n;i++){
            char c=s.charAt(i);
            int idx=(int)(c-'a');
            if(first[idx]==-1){
                first[idx]=i;
            }
            else{
                last[idx]=i;
            }
        }
        for(int i=0;i<26;i++){
            if(first[i]!=-1&&last[i]!=-1){
                boolean seen[]=new boolean[26];
                for(int j=first[i]+1;j<last[i];j++){
                    if(!seen[(int)(s.charAt(j)-'a')]){
                        ans++;
                        seen[(int)(s.charAt(j)-'a')]=true;
                    }
                }
            }
        }
        return ans;
    }
}
