class Solution {
    public int minimumDeletions(String s) {
        int aCount = 0;
        for (char c : s.toCharArray()) {
            if (c == 'a') aCount++;
        }

        int bCount = 0;
        int ans = aCount;

        for (char c : s.toCharArray()) {
            if (c == 'a') {
                aCount--;
            } else {
                bCount++;
            }
            ans = Math.min(ans, aCount + bCount);
        }

        return ans;
    }
}
