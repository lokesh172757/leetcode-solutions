class Solution {
    public List<String> summaryRanges(int[] nums) {
        List<String>ans=new ArrayList<>();
        if(nums.length==0){
            return ans;
        }
        int first=nums[0];
        for(int i=1;i<nums.length;i++){
            if(nums[i]-1!=nums[i-1]){
                int last=nums[i-1];
                if(first==last){
                    ans.add(first+"");
                }
                else{
                    ans.add(first+"->"+last);
                }
                first=nums[i];
            }
        }
        if(first==nums[nums.length-1]){
            ans.add(first+"");
        }
        else{
            ans.add(first+"->"+nums[nums.length-1]);
        }
        return ans;
    }
}
