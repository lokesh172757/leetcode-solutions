class Solution {
    public static int count(int grid[][],int i,int j,int m,int n){
        if(i<0||j<0||i>=m||j>=n||grid[i][j]==0){
            return 0;
        }
        grid[i][j]=0;
        return 1+count(grid,i+1,j,m,n)+count(grid,i-1,j,m,n)+count(grid,i,j+1,m,n)+count(grid,i,j-1,m,n);
    }
    public int maxAreaOfIsland(int[][] grid) {
        int ans=0,m=grid.length,n=grid[0].length;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(grid[i][j]==1){
                    ans=Math.max(ans,count(grid,i,j,m,n));
                }
            }
        }
        return ans;
    }
}
