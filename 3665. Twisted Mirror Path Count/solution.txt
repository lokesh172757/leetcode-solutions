class Solution {
    static int mod=(int)1e9+7;
    public static long find(int grid[][],int i,int j,int d,long dp[][]){
        if(i>=grid.length||j>=grid[0].length)return 0L;
        if(i==grid.length-1&&j==grid[0].length-1)return 1L;
        if(dp[i][j]!=-1)return dp[i][j];
        if(grid[i][j]==1){
            if(d==1)return find(grid,i,j+1,0,dp);
            return find(grid,i+1,j,1,dp);
        }
        return dp[i][j]=(find(grid,i,j+1,0,dp)+find(grid,i+1,j,1,dp))%mod;
    }
    public int uniquePaths(int[][] grid) {
        int m=grid.length,n=grid[0].length;
        long dp[][]=new long[m][n];
        for(long d[]:dp){
            Arrays.fill(d,-1);
        }
        return (int)find(grid,0,0,-1,dp);
    }
}
