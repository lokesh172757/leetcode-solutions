class Solution {
    public static int findLen(int i,HashMap<Integer,Integer>map,int len){
        int freq=0;
        if(map.containsKey(i))freq=map.get(i);
        if(freq==0)return 0;
        else if(freq==1)return len+1;
        int ans=findLen(i*i,map,len+2);
        if(ans==0)return len+1;
        return ans;
    }
    public int maximumLength(int[] nums) {
        int n=nums.length,countOne=0,max=0;
        HashMap<Integer,Integer>map=new HashMap<>();
        for(int num:nums){
            if(num==1)countOne++;
            else map.put(num,map.getOrDefault(num,0)+1);
            max=Math.max(num,max);
        }
        int ans=1;
        if(countOne%2==1)ans=countOne;
        else if(countOne>0)ans=countOne-1;
        if(max==1)return ans;
        int lim=(int)Math.sqrt(max);
        for(int i=2;i<=lim;i++){
            if(map.containsKey(i)&&map.get(i)>=2){
                ans=Math.max(ans,findLen(i*i,map,2));
            }
        }
        return ans;
    }
}
