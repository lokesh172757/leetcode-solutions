class Solution {
    public String maxSumOfSquares(int num, int sum) {
        StringBuilder sb=new StringBuilder("");
        int dig=9;
        while(num>0&&sum>0){
            if(sum>=dig){
                sum-=dig;
                sb.append((char)(dig+48));
                num--;
            }
            else dig--;
        }
        if(sum>0)return "";
        while(num>0){
            sb.append('0');
            num--;
        }
        return sb.toString();
    }
}
