import java.util.*;

class Solution {
    public static int help(int i, int j, int matrix[][], int dp[][]) {
        if (i >= matrix.length || j < 0 || j >= matrix[0].length) {
            return Integer.MAX_VALUE;
        }
        if (i == matrix.length - 1) {
            return matrix[i][j];
        }
        if (dp[i][j] != Integer.MAX_VALUE) return dp[i][j];

        int ans1 = help(i + 1, j + 1, matrix, dp);
        int ans2 = help(i + 1, j, matrix, dp);
        int ans3 = help(i + 1, j - 1, matrix, dp);

        return dp[i][j] = Math.min(ans1, Math.min(ans2, ans3)) + matrix[i][j];
    }

    public int minFallingPathSum(int[][] matrix) {
        int n = matrix.length;
        int m = matrix[0].length;
        int dp[][] = new int[n][m];

        for (int i = 0; i < n; i++) {
            Arrays.fill(dp[i], Integer.MAX_VALUE);
        }

        int ans = Integer.MAX_VALUE;
        for (int j = 0; j < m; j++) {
            ans = Math.min(ans, help(0, j, matrix, dp));
        }

        return ans;
    }
}
