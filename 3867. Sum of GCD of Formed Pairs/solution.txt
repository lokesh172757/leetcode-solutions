class Solution {
    public static int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return Math.abs(a);
    }
    public long gcdSum(int[] nums) {
        int n=nums.length,max=nums[0];
        long ans=0;
        int prefix[]=new int[n];
        for(int i=0;i<n;i++){
            max=Math.max(max,nums[i]);
            prefix[i]=gcd(max,nums[i]);
            
        }
        Arrays.sort(prefix);
        int i=0,j=n-1;
        while(i<j){
            ans+=gcd(prefix[i],prefix[j]);
            i++;
            j--;
        }
        return ans;
    }
}
