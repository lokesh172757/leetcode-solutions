class Solution {
    public int countPrefixSuffixPairs(String[] words) {
        int ans=0,n=words.length;
        for(int i=0;i<n;i++){
            int len1=words[i].length();
            for(int j=i+1;j<n;j++){
                int len2=words[j].length();
                if(len1>len2)continue;
                else if(words[i].equals(words[j].substring(0,len1))&&words[i].equals(words[j].substring(len2-len1)))ans++;
            }
        }
        return ans;
    }
}
