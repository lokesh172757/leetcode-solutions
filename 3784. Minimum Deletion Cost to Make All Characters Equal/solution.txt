class Solution {
    public long minCost(String s, int[] cost) {
        int n=cost.length;
        long sum=0;
        long arr[]=new long[26];
        for(int i=0;i<n;i++){
            char ch=s.charAt(i);
            arr[(int)(ch-'a')]+=cost[i];
            sum+=cost[i];
        }
        long ans=Long.MAX_VALUE;
        for(int i=0;i<26;i++){
            ans=Math.min(ans,sum-arr[i]);
        }
        return ans;
    }
}
