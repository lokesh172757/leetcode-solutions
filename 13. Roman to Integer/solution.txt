class Solution {
    static int it[]={1000,900,500,400,100,90,50,40,10,9,5,4,1};
    static String roman[]={"M","CM","D","CD","C","XC","L","XL","X","IX","V","IV","I"};
    static HashMap<String,Integer>map=new HashMap<>();
    boolean mapCreated=false;
    public int romanToInt(String s) {
        if(!mapCreated){
            for(int i=0;i<it.length;i++){
                map.put(roman[i],it[i]);
            }
            mapCreated=true;
        }
        int ans=0;
        for(int i=0;i<s.length();i++){
            String ch=s.substring(i,i+1);
            if(i<s.length()-1){
                String c=s.substring(i+1,i+2);
                int a=map.get(ch),b=map.get(c);
                if(b>a){
                    ans+=b-a;
                    i++;
                }
                else ans+=a;
            }
            else ans+=map.get(ch);
        }
        return ans;
    }
}
