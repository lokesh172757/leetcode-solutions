class Solution {
    public int[] findRightInterval(int[][] intervals) {
        int n=intervals.length;
        int ans[]=new int[n];
        int arr[][]=new int[n][3];
        for(int i=0;i<n;i++){
            arr[i][0]=intervals[i][0];
            arr[i][1]=intervals[i][1];
            arr[i][2]=i;
        }
        Arrays.sort(arr,Comparator.comparingInt(o->o[0]));
        for(int i=0;i<n;i++){
            boolean find =false;
            for(int j=i;j<n;j++){
                if(arr[j][0]>=arr[i][1]){
                    find =true;
                    ans[arr[i][2]]=arr[j][2];
                    break;
                }
            }
            if(!find)ans[arr[i][2]]=-1;
        }
        return ans;
    }
}
