class CustomStack {
    int[][] stack;
    int top;
    public CustomStack(int maxSize) {
        stack = new int[maxSize][2];
        top = -1;
    }
    public void push(int x) {
        if (top == stack.length - 1) {
            return;
        }
        top++;
        stack[top][0] = x;
    }
    public int pop() {
        if (top == -1) {
            return -1;
        }
        int ans = stack[top][0] + stack[top][1];
        if (top > 0) {
            stack[top - 1][1] += stack[top][1];
        }
        stack[top][1] = 0;
        top--;
        return ans;
    }
    public void increment(int k, int val) {
        if (top == -1) {
            return;
        }
        int idx = Math.min(k - 1, top);
        stack[idx][1] += val;
    }
}
