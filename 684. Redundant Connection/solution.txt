class Solution {
    public static void union(int a,int b,int par[],int rank[]){
        int parA=find(a,par),parB=find(b,par);
        if(rank[parA]<rank[parB]){
            par[parA]=parB;
        }
        else if(rank[parA]>rank[parB]){
            par[parB]=parA;
        }
        else{
            par[parA]=parB;
            rank[parA]++;
        }
    }
    public static int find(int x,int par[]){
        if(par[x]!=x){
            par[x]=find(par[x],par);
        }
        return par[x];
    }
    public int[] findRedundantConnection(int[][] edges) {
        int ans=-1,n=edges.length;
        int par[]=new int[n+1];
        int rank[]=new int[n+1];
        for(int i=1;i<=n;i++){
            par[i]=i;
        }
        for(int i=0;i<n;i++){
            int a=edges[i][0],b=edges[i][1];;
            if(find(a,par)==find(b,par)){
                ans=i;
            }
            else{
                union(a,b,par,rank);
            }
        }
        return edges[ans];
    }
}
