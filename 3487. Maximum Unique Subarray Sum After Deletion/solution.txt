class Solution {
    public int maxSum(int[] nums) {
        int max=Integer.MIN_VALUE;
        int arr[]=new int[101];
        int sum=0;
        for (int num : nums) {
            if(num>max){
                max=num;
            }
            if(num>0&&arr[num]==0){
                arr[num]=1;
                sum+=num;
            }
        }
        if(max<0){
            return max;
        }
        return sum;
    }
}
