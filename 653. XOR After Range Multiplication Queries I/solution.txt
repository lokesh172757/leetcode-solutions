class Solution {
    static int mod=(int)1e9+7;
    public int xorAfterQueries(int[] nums, int[][] queries) {
        int n=nums.length;
        for(int i=0;i<queries.length;i++){
            int l=queries[i][0],r=queries[i][1],k=queries[i][2],v=queries[i][3];
            while(l<=r){
                nums[l]=(int)(((long)nums[l]*v)%mod);
                l+=k;
            }
        }
        int ans=nums[0];
        for(int i=1;i<n;i++){
            ans^=nums[i];
        }
        return ans;
    }
}
