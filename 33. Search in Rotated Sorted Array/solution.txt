class Solution {
    public int search(int[] nums, int target) {
        int i=0,j=nums.length-1;
        while(i<=j){
            int mid=i+(j-i)/2;
            if(nums[mid]==target)return mid;
            //line 1
            if(nums[mid]>=nums[i]&&nums[mid]>nums[j]){
                //case a
                if(target>=nums[i]&&target<nums[mid]){
                    j=mid-1;
                }
                else i++;//case b
            }
            else{//line 2
                // case c
                if(target<=nums[j]&&target>nums[mid])i=mid+1;
                else j--;//case d
            }
        }
        return -1;
    }
}
