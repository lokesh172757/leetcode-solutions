class Solution {
    public int robotSim(int[] commands, int[][] obstacles) {
        int x = 0, y = 0, n = commands.length, max = 0;

        HashSet<String> set = new HashSet<>();
        for (int[] ar : obstacles) {
            set.add(ar[0] + "," + ar[1]);
        }

        char face = 'N';

        for (int i = 0; i < n; i++) {
            int command = commands[i];

            if (command == -2) {
                if (face == 'N') face = 'W';
                else if (face == 'W') face = 'S';
                else if (face == 'S') face = 'E';
                else face = 'N';
            }
            else if (command == -1) {
                if (face == 'N') face = 'E';
                else if (face == 'E') face = 'S';
                else if (face == 'S') face = 'W';
                else face = 'N';
            }
            else {
                if (face == 'N') {
                    while (command-- > 0) {
                        if (set.contains(x + "," + (y + 1))) break;
                        y++;
                        max = Math.max(max, x * x + y * y);
                    }
                }
                else if (face == 'S') {
                    while (command-- > 0) {
                        if (set.contains(x + "," + (y - 1))) break;
                        y--;
                        max = Math.max(max, x * x + y * y);
                    }
                }
                else if (face == 'E') {
                    while (command-- > 0) {
                        if (set.contains((x + 1) + "," + y)) break;
                        x++;
                        max = Math.max(max, x * x + y * y);
                    }
                }
                else {
                    while (command-- > 0) {
                        if (set.contains((x - 1) + "," + y)) break;
                        x--;
                        max = Math.max(max, x * x + y * y);
                    }
                }
            }
        }

        return max;
    }
}
