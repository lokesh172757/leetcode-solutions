class Solution {
    public boolean findRotation(int[][] mat, int[][] target) {
        int n=mat.length;
        int arr[][]=new int[n][n];
        boolean equal=true;
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                if(mat[i][j]!=target[i][j]){
                    equal=false;
                    break;
                }
            }
        }
        if(equal)return true;
        for(int k=1;k<4;k++){
            equal=true;
            for(int i=0;i<n;i++){
                for(int j=0;j<n;j++){
                    arr[j][n-i-1]=mat[i][j];
                }
            }
            for(int i=0;i<n;i++){
                for(int j=0;j<n;j++){
                    if(arr[i][j]!=target[i][j]){
                        equal=false;
                        break;
                    }
                }
            }
            if(equal)return true;
            int temp[][]=arr;
            arr=mat;
            mat=temp;
        }
        return false;
    }
}
