class Solution {
    public int[] productQueries(int n, int[][] queries) {
        int l[]=new int[32],k=0;
        for(int i=0;i<32;i++){
            if((n&(1<<i))!=0){
                l[k]=1<<i;
                k++;
            }
        }
        int mod=(int)1e9+7;
        int ans[]=new int[queries.length];
        for(int i=0;i<queries.length;i++){
            int st=queries[i][0],e=queries[i][1];
            long prod=1;
            for(int j=st;j<=e;j++){
                prod=(l[j]*prod)%mod;

            }
            ans[i]=(int)prod;
        }
        return ans;
    }
}
