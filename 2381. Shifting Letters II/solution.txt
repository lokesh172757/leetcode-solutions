class Solution {
    public String shiftingLetters(String s, int[][] shifts) {
        int n = s.length();
        int[] diff = new int[n + 1];
        for (int[] sh : shifts) {
            int l = sh[0], r = sh[1];
            int val = (sh[2] == 1) ? 1 : -1;
            diff[l] += val;
            diff[r + 1] -= val;
        }

        StringBuilder sb = new StringBuilder();
        int shift = 0;

        for (int i = 0; i < n; i++) {
            shift += diff[i];

            int normalized = ((s.charAt(i) - 'a') + shift) % 26;
            if (normalized < 0) normalized += 26;

            sb.append((char) (normalized + 'a'));
        }

        return sb.toString();
    }
}
